from pathlib import Path

def validate_topic_id(topic):
    import re
    if not re.fullmatch(r"[A-Za-z0-9_-]{3,80}", str(topic or "")):
        raise ValueError("Invalid topic ID; path separators and traversal are not allowed.")

def latest_draft(topic):
    validate_topic_id(topic)
    base = ROOT / "production" / "runs" / topic / "drafts"
    if not base.is_dir():
        raise RuntimeError(f"Draft directory not found for topic: {topic}")
    candidates = []
    for p in base.iterdir():
        if p.is_dir() and (p / "creative_package.json").is_file():
            fact = p / "automated_fact_check_v2_report.json"
            if fact.is_file():
                try:
                    if json.loads(fact.read_text(encoding="utf-8-sig")).get("status") == "PASS":
                        candidates.append(p)
                except Exception:
                    pass
    if not candidates:
        raise RuntimeError("No draft with creative package and passing fact-check report.")
    return max(candidates, key=lambda p: p.stat().st_mtime)

def digest(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def seed_for(text, attempt):
    raw = (text + "|" + str(attempt)).encode("utf-8")
    return int(hashlib.sha256(raw).hexdigest()[:16], 16)

def make_artwork(task, output, attempt=0, run_salt=""):
    width, height = task["width"], task["height"]
    rng = random.Random(seed_for(task["asset_id"] + task["visual_direction"] + "|" + run_salt, attempt))

    palettes = [
        ((8, 15, 35), (24, 55, 86), (13, 112, 124)),
        ((18, 12, 35), (72, 35, 83), (169, 91, 82)),
        ((7, 24, 29), (18, 75, 78), (94, 137, 111)),
        ((22, 18, 15), (83, 57, 36), (169, 119, 66)),
        ((12, 17, 29), (39, 54, 83), (103, 89, 140)),
    ]
    c1, c2, accent = rng.choice(palettes)

    # Cinematic gradient background.
    gradient = Image.linear_gradient("L").resize((width, height))
    background = Image.new("RGB", (width, height), c1)
    top = Image.new("RGB", (width, height), c2)
    background = Image.composite(top, background, gradient)

    # Soft light source.
    glow = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    gd = ImageDraw.Draw(glow)
    gx = rng.randint(width // 4, width * 3 // 4)
    gy = rng.randint(height // 4, height * 3 // 4)
    radius = int(min(width, height) * rng.uniform(.20, .42))
    gd.ellipse((gx-radius, gy-radius, gx+radius, gy+radius),
               fill=(*accent, 115))
    glow = glow.filter(ImageFilter.GaussianBlur(radius * .55))
    background = Image.alpha_composite(background.convert("RGBA"), glow)

    # Original geometric documentary composition.
    layer = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    draw = ImageDraw.Draw(layer)
    short = min(width, height)

    # Fine grid and perspective lines.
    spacing = max(55, short // 13)
    for x in range(0, width, spacing):
        draw.line((x, 0, x, height), fill=(210, 225, 235, 13), width=1)
    for y in range(0, height, spacing):
        draw.line((0, y, width, y), fill=(210, 225, 235, 13), width=1)

    # Layered rings / orbital geometry.
    cx = rng.randint(width // 3, width * 2 // 3)
    cy = rng.randint(height // 3, height * 2 // 3)
    for k in range(5):
        r = int(short * (.12 + k * .075))
        draw.ellipse((cx-r, cy-r, cx+r, cy+r),
                     outline=(*accent, max(18, 90-k*12)),
                     width=max(2, short // 500))

    # Directional beams.
    for _ in range(16):
        x1 = rng.randint(-width // 3, width)
        y1 = rng.randint(-height // 3, height)
        x2 = x1 + rng.randint(-width // 2, width // 2)
        y2 = y1 + rng.randint(-height // 2, height // 2)
        draw.line((x1, y1, x2, y2),
                  fill=(*accent, rng.randint(12, 45)),
                  width=rng.randint(1, max(2, short // 300)))

    # Abstract focal forms.
    for _ in range(12):
        x = rng.randint(0, width)
        y = rng.randint(0, height)
        r = rng.randint(short // 45, short // 10)
        col = rng.choice([(*accent, 28), (220, 230, 240, 18), (4, 8, 18, 50)])
        draw.ellipse((x-r, y-r, x+r, y+r), fill=col,
                     outline=(*accent, 65), width=max(1, short // 600))

    # Unique scene-specific diagonal motif.
    points = [
        (rng.randint(0, width), rng.randint(0, height))
        for _ in range(7)
    ]
    draw.line(points, fill=(*accent, 150), width=max(3, short // 220),
              joint="curve")

    # Scene-aware visual motif. This is deterministic and uses only Pillow
    # primitives; it is not represented as photorealistic AI imagery.
    scene_text = " ".join([
        str(task.get("visual_direction", "")),
        str(task.get("narration_excerpt", "")),
        str(task.get("section", "")),
    ]).lower()
    cx, cy = width // 2, height // 2
    unit = max(12, short // 35)
    ink = (*accent, 220)
    pale = (225, 235, 245, 190)

    def line(coords, fill=ink, w=None):
        draw.line(coords, fill=fill, width=w or max(3, unit // 5), joint="curve")

    def ellipse(box, outline=ink, w=None, fill=None):
        draw.ellipse(box, outline=outline, width=w or max(3, unit // 5), fill=fill)

    if any(k in scene_text for k in ("money", "wealth", "finance", "cash", "debt", "invest", "rich", "poor", "price", "income")):
        # Coin stack + rising financial trajectory.
        for n in range(4):
            y = cy + unit * (2 - n)
            draw.rounded_rectangle((cx-unit*3, y, cx-unit, y+unit//2),
                radius=unit//5, outline=pale, width=max(3, unit//6), fill=(20, 35, 55, 120))
        line([(cx, cy+unit*2), (cx+unit, cy+unit), (cx+unit*2, cy+unit//2), (cx+unit*3, cy-unit*2)], w=max(5,unit//4))
        line([(cx+unit*2,cy-unit*2),(cx+unit*3,cy-unit*2),(cx+unit*3,cy-unit)], w=max(5,unit//4))
    elif any(k in scene_text for k in ("brain", "psychology", "mind", "thought", "emotion", "memory", "overthink", "behavior", "decision")):
        # Abstract brain hemispheres with neural connections.
        ellipse((cx-unit*3,cy-unit*3,cx,cy+unit*3), pale, max(4,unit//4))
        ellipse((cx,cy-unit*3,cx+unit*3,cy+unit*3), pale, max(4,unit//4))
        line([(cx,cy-unit*2),(cx-unit,cy-unit),(cx-unit*2,cy),(cx-unit,cy+unit),(cx,cy+unit*2)],w=max(3,unit//5))
        line([(cx,cy-unit*2),(cx+unit,cy-unit),(cx+unit*2,cy),(cx+unit,cy+unit),(cx,cy+unit*2)],w=max(3,unit//5))
        for dx,dy in [(-2,-1),(-1,1),(1,-1),(2,1)]:
            ellipse((cx+dx*unit-unit//5,cy+dy*unit-unit//5,cx+dx*unit+unit//5,cy+dy*unit+unit//5),ink,fill=ink)
    elif any(k in scene_text for k in ("power", "status", "hierarchy", "influence", "leader", "control", "authority", "dominance")):
        # Tiered hierarchy / influence pyramid.
        draw.polygon([(cx,cy-unit*3),(cx-unit*3,cy+unit*2),(cx+unit*3,cy+unit*2)], outline=pale, fill=(20,35,55,100))
        line([(cx-unit*2,cy+unit//3),(cx+unit*2,cy+unit//3)],pale)
        line([(cx-unit*5//2,cy+unit),(cx+unit*5//2,cy+unit)],pale)
        for dx,dy in [(0,-unit*2),(-unit,unit),(unit,unit)]:
            ellipse((cx+dx-unit//4,cy+dy-unit//4,cx+dx+unit//4,cy+dy+unit//4),ink,fill=ink)
    elif any(k in scene_text for k in ("time", "habit", "routine", "discipline", "focus", "productivity", "morning", "clock")):
        # Clock face with deliberate hands.
        ellipse((cx-unit*3,cy-unit*3,cx+unit*3,cy+unit*3),pale,max(5,unit//4))
        line([(cx,cy),(cx,cy-unit*2)],pale,max(5,unit//4))
        line([(cx,cy),(cx+unit*2,cy+unit)],ink,max(5,unit//4))
        ellipse((cx-unit//4,cy-unit//4,cx+unit//4,cy+unit//4),ink,fill=ink)
    elif any(k in scene_text for k in ("social", "phone", "screen", "notification", "internet", "media", "scroll")):
        # Smartphone silhouette with notification.
        draw.rounded_rectangle((cx-unit*2,cy-unit*3,cx+unit*2,cy+unit*3),
            radius=unit//2, outline=pale, width=max(5,unit//4), fill=(10,20,35,130))
        line([(cx-unit,cy-unit*2),(cx+unit,cy-unit*2)],pale)
        ellipse((cx-unit//4,cy+unit*2,cx+unit//4,cy+unit*2+unit//2),pale)
        ellipse((cx+unit,cy-unit*3-unit//3,cx+unit*2,cy-unit*2),ink,fill=ink)

    composed = Image.alpha_composite(background, layer)

    # Subtle vignette.
    mask = Image.new("L", (width, height), 0)
    md = ImageDraw.Draw(mask)
    md.ellipse((-width//4, -height//4, width*5//4, height*5//4),
               fill=210)
    mask = mask.filter(ImageFilter.GaussianBlur(short // 5))
    dark = Image.new("RGBA", (width, height), (0, 0, 0, 90))
    dark.putalpha(ImageChops.invert(mask).point(lambda p: int(p * .65)))
    composed = Image.alpha_composite(composed, dark)

    composed.convert("RGB").save(output, "JPEG", quality=94, optimize=True)
    return output

def run(topic):
    validate_topic_id(topic)
    from research_approval_guard_v1 import require_research_approval
    require_research_approval(topic)
    draft = latest_draft(topic)
    creative_path = draft / "creative_package.json"
    creative = json.loads(creative_path.read_text(encoding="utf-8-sig"))
    tasks = collect_tasks(creative)
    if not tasks:
        raise RuntimeError("Creative package contains no visual tasks.")

    stamp = datetime.now().astimezone().strftime("%Y%m%d_%H%M%S")
    package_dir = draft / f"automated_visual_assets_v1_{stamp}"
    assets_dir = package_dir / "assets"
    assets_dir.mkdir(parents=True, exist_ok=False)

    results, hashes = [], set()
    for task in tasks:
        filename = task["asset_id"] + ".jpg"
        path = assets_dir / filename
        errors = []

        # Bounded local regeneration: maximum two attempts.
        selected_attempt = 0
        for attempt in range(2):
            selected_attempt = attempt
            make_artwork(task, path, attempt, run_salt=stamp)
            try:
                with Image.open(path) as im:
                    im.verify()
                with Image.open(path) as im:
                    valid = im.size == (task["width"], task["height"])
                file_hash = digest(path)
                if valid and file_hash not in hashes:
                    break
                errors.append("DIMENSION_OR_DUPLICATE_RETRY")
            except Exception as exc:
                errors.append("IMAGE_DECODE_RETRY:" + str(exc))
        else:
            results.append({
                **task, "asset_path": "assets/" + filename,
                "status": "REJECTED", "errors": errors
            })
            continue

        hashes.add(file_hash)
        results.append({
            **task,
            "asset_path": "assets/" + filename,
            "status": "AUTOMATED_TECHNICAL_PASS",
            "sha256": file_hash,
            "provenance": {
                 "source_type": "local_procedural_original",
                "creator": "Project-authored local visual renderer",
                "source_code_reference": "production/automated_visual_engine_v1.py",
                "rights_basis": "Generated from project-authored Pillow drawing primitives; no external image inputs.",
                "commercial_use_confirmed": True,
                "renderer": "project-authored Pillow drawing pipeline",
                "renderer_version": VERSION,
                "external_images_used": False,
                "stock_media_used": False,
                "third_party_fonts_used": False,
                "generation_seed": seed_for(task["asset_id"] + task["visual_direction"] + "|" + stamp, selected_attempt),
                "generation_run_id": stamp,
            },
            "quality": {
                "width": task["width"], "height": task["height"],
                "format": "JPEG", "decode": "PASS",
                "exact_duplicate": False,
            },
            "errors": errors,
        })

    passed = sum(x["status"] == "AUTOMATED_TECHNICAL_PASS" for x in results)
    failed = len(results) - passed
    manifest = {
        "version": VERSION,
        "topic_id": topic,
        "created_at": datetime.now().astimezone().isoformat(),
        "draft_directory": str(draft),
        "package_directory": str(package_dir),
        "generation_method": "local_procedural_original",
        "external_media_used": False,
        "total_assets": len(results),
        "automated_technical_pass": passed,
        "rejected": failed,
        "assets": results,
        "automated_visual_gate": "PASS" if failed == 0 else "HOLD",
        "limitations": [
            "These are original procedural graphics, not photorealistic AI images.",
            "Automated checks validate dimensions, file integrity and exact duplicates.",
            "Visual semantic relevance is not independently understood by a vision model.",
            "No external stock or AI provider was used; this is not a legal opinion on copyright.",
            "Technical visual pass does not authorize rendering, production or publication.",
        ],
        "approval": {
            "automated_technical_gate_pass": failed == 0,
            "production_approved": False,
            "render_authorized": False,
            "upload_authorized": False,
            "public_upload_authorized": False,
        },
    }
    manifest_path = package_dir / "asset_manifest.json"
    manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")
    report_path = package_dir / "automated_visual_gate_report.json"
    report_path.write_text(json.dumps({
        "status": manifest["automated_visual_gate"],
        "total": len(results), "passed": passed, "rejected": failed,
        "manifest": str(manifest_path),
        "production_approved": False,

    print("=" * 60)

def sha256_file(path):
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def normalize_text(value):
    value = re.sub(r"```.*?```", " ", value, flags=re.S)
    value = re.sub(r"^\s*#{1,6}\s+.*$", " ", value, flags=re.M)
    value = re.sub(r"https?://\S+", " ", value)
    value = value.lower()
    value = re.sub(r"[^a-z0-9\u0900-\u097f]+", " ", value)

    return " ".join(
        word for word in value.split()
        if len(word) > 2 and word not in IGNORED_WORDS
    )


def make_shingles(text, size=5):
    words = text.split()
    if len(words) < size:
        return {" ".join(words)} if words else set()

    return {
        " ".join(words[index:index + size])
        for index in range(len(words) - size + 1)
    }


def jaccard_similarity(first, second):
    if not first or not second:
        return 0.0

    union = first | second
    return len(first & second) / len(union) if union else 0.0


def average_hash(path):
    with Image.open(path) as image:
        image = image.convert("L").resize((16, 16))

        try:
            pixels = list(image.get_flattened_data())
        except AttributeError:
            pixels = list(image.getdata())

    average = sum(pixels) / len(pixels)
    value = 0

    for pixel in pixels:
        value = (value << 1) | int(pixel >= average)

    return value


def hash_similarity(first, second):
    return 1.0 - ((first ^ second).bit_count() / 256)


def topic_id_from_draft(draft):
    # runs/<topic_id>/drafts/<draft_id>
    try:
        return draft.resolve().parents[1].name
    except IndexError:
        return None


def topic_id_from_script(script_path):
    # runs/<topic_id>/drafts/<draft_id>/<script>.md
    try:
        return script_path.resolve().parents[2].name
    except IndexError:
        return None


def load_script(path):
    if not path.is_file():
        return ""

    return normalize_text(
        path.read_text(encoding="utf-8", errors="replace")
    )


def collect_other_topic_scripts(runs_root, current_draft, kind):
    filename = SCRIPT_NAMES[kind]
    current_topic = topic_id_from_draft(current_draft)
    results = []

    if not runs_root.is_dir():
        return results

    for path in runs_root.rglob(filename):
        try:
            resolved = path.resolve()

            if current_draft.resolve() in resolved.parents:
                continue

            old_topic = topic_id_from_script(resolved)

            if not old_topic or old_topic == current_topic:
                continue

            content = load_script(resolved)

            if content:
                results.append({
                    "path": resolved,
                    "topic_id": old_topic,
                    "content": content,
                })

        except (OSError, RuntimeError):
            continue

    return results


def classify_script_corpus(previous_scripts, comparisons, jaccard_threshold, sequence_threshold):
    """Classify project-local script comparison; first script seeds the corpus."""
    if not previous_scripts:
        return "BASELINE_SEED_NO_PRIOR_TOPIC_SCRIPTS"
    if any(
        item["jaccard_similarity"] >= jaccard_threshold
        or item["sequence_similarity"] >= sequence_threshold
        for item in comparisons
    ):
        return "HOLD_POSSIBLE_REUSE"
    return "COMPARISON_CLEAR"


def validate_provenance(asset):
    provenance = asset.get("provenance") or {}

    failures = []
    holds = []
    source_type = str(
        provenance.get("source_type", "")
    ).strip().lower()

    if source_type == "original":
        if not provenance.get("creator"):
            holds.append("ORIGINAL_CREATOR_MISSING")

        if provenance.get("rights_confirmation") is not True:
            holds.append("ORIGINAL_RIGHTS_NOT_CONFIRMED")

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("ORIGINAL_COMMERCIAL_USE_NOT_CONFIRMED")

        if not (
            provenance.get("source_url")
            or provenance.get("source_code_reference")
            or provenance.get("generation_prompt_used")
        ):
            holds.append("ORIGINAL_SOURCE_TRACEABILITY_MISSING")

    elif source_type == "ai_generated":
        required = (
            "provider",
            "model",
            "terms_reference",
            "terms_checked_at",
            "generation_prompt_used",
        )

        missing = [
            field for field in required
            if not provenance.get(field)
        ]

        if missing:
            holds.append(
                "AI_PROVENANCE_MISSING_" + "_".join(missing)
            )

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("AI_COMMERCIAL_USE_NOT_CONFIRMED")

        if provenance.get("rights_confirmation") is not True:
            holds.append("AI_RIGHTS_NOT_CONFIRMED")

    elif source_type == "local_procedural_original":
        required = ("creator", "source_code_reference", "rights_basis")
        missing = [field for field in required if not provenance.get(field)]
        if missing:
            holds.append("LOCAL_PROCEDURAL_PROVENANCE_MISSING_" + "_".join(missing))
        if provenance.get("external_images_used") is not False:
            holds.append("LOCAL_PROCEDURAL_EXTERNAL_INPUTS_NOT_EXCLUDED")
        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("LOCAL_PROCEDURAL_COMMERCIAL_USE_NOT_CONFIRMED")

    elif source_type == "stock":
        if not provenance.get("source_url"):
            holds.append("STOCK_SOURCE_URL_MISSING")

        if not provenance.get("license_name"):
            holds.append("STOCK_LICENSE_MISSING")

        if not provenance.get("license_url"):
            holds.append("STOCK_LICENSE_REFERENCE_MISSING")

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("STOCK_COMMERCIAL_USE_NOT_CONFIRMED")

    else:
        holds.append("UNKNOWN_OR_UNDECLARED_SOURCE_TYPE")

    return failures, holds


def validate_asset_path(manifest_dir, relative_path):
    if not relative_path:
        return None, "ASSET_PATH_MISSING"

    candidate = (manifest_dir / relative_path).resolve()

    try:
        candidate.relative_to(manifest_dir.resolve())
    except ValueError:
        return None, "ASSET_PATH_ESCAPES_PACKAGE"

    if not candidate.is_file():
        return None, "ASSET_FILE_MISSING"

    return candidate, None

    if not assets:
        hard_failures.append("MANIFEST_ASSET_LIST_EMPTY")

                details["sha256"] = file_hash

                exact_hash_groups.setdefault(
                    file_hash, []
                ).append(asset_id)

                with Image.open(image_path) as image:

                with Image.open(image_path) as image:

                or sequence_score >= args.script_sequence_threshold

    else:

        for result in asset_results

def load_json(path: Path) -> dict:
    if not path.is_file():
        raise FileNotFoundError(str(path))
    data = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(data, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return data


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def inside(child: Path, parent: Path) -> bool:
    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except ValueError:
        return False


def add(items: list, code: str, detail: str = ""):
    items.append({"code": code, "detail": detail})


def count_narration_words(text: str) -> int:
    sections = []
    heading = None
    lines = []
    for line in text.splitlines():
        if line.startswith("## "):
            if heading and heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
                sections.extend(lines)
            heading = line[3:].strip()
            lines = []
            continue
        value = line.strip()
        if not heading:
            continue
        if value.lower().startswith("claim mapping:") or value.startswith("- "):
            continue
        if value:
            lines.append(value)
    if heading and heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
        sections.extend(lines)
    return len(re.findall(r"\b[\w’'-]+\b", " ".join(sections)))

                })

                    asset_path = None

            if asset_path is None or not asset_path.is_file():
                reasons.append("ASSET_FILE_MISSING")
            else:
                asset_mtimes.append(asset_path.stat().st_mtime)

                try:
                    from PIL import Image
                    with Image.open(asset_path) as im:
                        width, height = im.size
                        im.verify()

                    fmt = str(asset.get("format", "")).lower()
                    if fmt in ("portrait", "vertical"):
                        minimum = (1080, 1920)
                    else:
                        minimum = (1920, 1080)

                    if width < minimum[0] or height < minimum[1]:
                        reasons.append(
                            f"BELOW_PREMIUM_DIMENSIONS_{minimum[0]}x{minimum[1]}"
                            f"_ACTUAL_{width}x{height}"
                        )

                    quality = asset.get("quality") or {}
                    expected_hash = str(
                        quality.get("sha256") or asset.get("sha256") or ""
                    ).lower()
                    actual_hash = sha256(asset_path)

                    if not expected_hash:
                        reasons.append("ASSET_SHA256_MISSING")
                    elif expected_hash != actual_hash:
                        reasons.append("ASSET_SHA256_MISMATCH")

                    if actual_hash in seen_hashes:
                        reasons.append("EXACT_DUPLICATE_ASSET")
                    else:
                        seen_hashes[actual_hash] = aid

                except Exception as exc:
                    reasons.append("IMAGE_READ_OR_VALIDATION_ERROR:" + str(exc))

            provenance = asset.get("provenance") or {}
            source_type = str(provenance.get("source_type", "")).lower()
            creator = str(provenance.get("creator", "")).strip()
            prompt_used = str(
                provenance.get("generation_prompt_used")
                or asset.get("visual_direction", "")
            ).strip()

            supported_types = (
                "original", "ai_generated", "licensed", "local_procedural_original"
            )
            if source_type not in supported_types:
                reasons.append("SOURCE_TYPE_UNKNOWN_OR_UNSUPPORTED")
            if not creator:
                reasons.append("CREATOR_NOT_DECLARED")
            if not prompt_used and source_type in (
                "original", "ai_generated", "local_procedural_original"
            ):
                reasons.append("GENERATION_PROMPT_PROVENANCE_MISSING")
            rights_confirmed = (
                provenance.get("rights_confirmation") is True
                or (
                    source_type == "local_procedural_original"
                    and bool(provenance.get("rights_basis"))
                )
            )
            if not rights_confirmed:
                reasons.append("RIGHTS_NOT_CONFIRMED")
            if provenance.get("commercial_use_confirmed") is not True:
                reasons.append("COMMERCIAL_USE_NOT_CONFIRMED")

            if source_type == "licensed":
                if not provenance.get("license_name") or not provenance.get("license_url"):
                    reasons.append("LICENSE_DETAILS_MISSING")

            if reasons:
                asset_fail_count += 1
                regeneration.append({
                    "asset_id": aid,
                    "asset_path": rel,
                    "reasons": sorted(set(reasons)),
                })

        check(
            "visual_asset_files_and_provenance",
            asset_fail_count == 0,
            f"{len(asset_records)} assets; {asset_fail_count} require regeneration"
        )

    # 7. Originality report
    originality = None
    try:
        originality = load_json(originality_path)
        identity_ok = (
            originality.get("topic_id_from_path") == topic_id
            and originality.get("topic_id_from_manifest") == topic_id
            and originality.get("topic_id_match") is True
        )
        check("originality_report_identity", identity_ok)
        if not identity_ok:
            add(blockers, "ORIGINALITY_REPORT_TOPIC_MISMATCH")

        gate_ok = (
            originality.get("gate_version") == "1.3.0"
            and originality.get("status") == "PASS_ORIGINALITY_GATE"
            and originality.get("gate_pass") is True
        )
        check(
            "premium_originality_gate_v1_2",
            gate_ok,
            originality.get("status", "MISSING")
        )
        if not gate_ok:
            add(
                regeneration,
                "PREMIUM_ORIGINALITY_GATE_FAILED",
                str(originality.get("status"))
            )

        if originality.get("rights_hold_asset_ids"):
            add(
                regeneration,
                "ORIGINALITY_RIGHTS_HOLDS",
                str(len(originality["rights_hold_asset_ids"])) + " assets"
            )

        if originality.get("script_holds"):
            add(
                blockers,
                "SCRIPT_NOVELTY_CORPUS_ORIGINALITY_HOLD",
                str(originality.get("script_holds"))
            )

    except Exception as exc:
        add(blockers, "ORIGINALITY_REPORT_MISSING_OR_INVALID", str(exc))
        check("premium_originality_gate_v1_2", False, str(exc))

    # 8. Originality report freshness against visual package and assets
    if manifest_path and originality_path.exists():
        source_times = [manifest_path.stat().st_mtime] + asset_mtimes
        fresh = bool(source_times) and originality_path.stat().st_mtime >= max(source_times)
        check("originality_report_freshness", fresh)
        if not fresh:
            add(blockers, "ORIGINALITY_REPORT_STALE")

    # 9. Check for an actual render implementation.
    # FFmpeg being installed on Windows is not proof of a project render engine.
    render_candidates = []
    production_root = Path("production")
    if production_root.is_dir():
        for p in production_root.rglob("*.py"):
            if p.name == Path(__file__).name:
                continue
            if re.search(r"render|video_engine", p.name, re.I):
                render_candidates.append(str(p))

    render_engine_present = bool(render_candidates)
    check(
        "project_render_engine_exists",
        render_engine_present,
        "; ".join(render_candidates) if render_candidates else "No render engine module found"
    )
    if not render_engine_present:
        add(blockers, "RENDER_ENGINE_NOT_IMPLEMENTED")

    # 10. Status: fail closed
    if rejections:
        status = "REJECTED"
    elif regeneration:
        status = "REGENERATE_REQUIRED"
    elif blockers:
        status = "BLOCKED"
    else:
        status = "PASS_TO_RENDER"

    report = {
        "gate_version": VERSION,
        "created_at": now,
        "topic_id": topic_id,
        "draft_directory": str(draft),
        "status": status,
        "checks_passed": sum(1 for x in checks if x["status"] == "PASS"),
        "checks_total": len(checks),
        "checks": checks,
        "blockers": blockers,
        "regeneration_queue": regeneration,
        "rejections": rejections,
        "warnings": warnings,
        "visual_asset_count": len(asset_records),
        "visual_assets_requiring_regeneration": asset_fail_count,
        "render_engine_detected": render_engine_present,
        "production_approved": False,
        "render_authorized": False,
        "upload_authorized": False,
        "public_upload_authorization": False,
        "limitations": [
            "This gate checks report consistency, file integrity, dimensions and declared provenance.",
            "It does not independently prove scientific truth, semantic originality or visual suitability.",
            "Rights declarations are not independently verified against provider terms.",
            "PASS_TO_RENDER is a readiness result only; it does not grant render or upload authorization.",
            "No source manifest, source asset, research file or approval flag is modified.",
        ],
    }

    report_path = draft / "central_production_gate_v1_report.json"
    report_path.write_text(
        json.dumps(report, indent=2, ensure_ascii=False),
        encoding="utf-8",
    )

    print("=" * 68)
    print("CENTRAL PRODUCTION GATE", VERSION)
    print("=" * 68)
    print("TOPIC:", topic_id)
    print("STATUS:", status)
    print("CHECKS:", report["checks_passed"], "/", report["checks_total"])
    print("VISUAL ASSETS:", len(asset_records))
    print("ASSETS REQUIRING REGENERATION:", asset_fail_count)
    print("BLOCKERS:", len(blockers))
    print("REGENERATION ITEMS:", len(regeneration))
    print("REJECTIONS:", len(rejections))
    print("RENDER ENGINE:", render_engine_present)
    print("PRODUCTION APPROVED: False")
    print("RENDER AUTHORIZED: False")
    print("UPLOAD AUTHORIZED: False")
    print("REPORT:", report_path)

    if regeneration:
        print("\nREGENERATION QUEUE:")
        for item in regeneration[:20]:
            if isinstance(item, dict):
                print("-", item.get("asset_id", item.get("code")), item.get("reasons", item.get("detail", "")))
            else:
                print("-", item)

    if blockers:
        print("\nBLOCKERS:")
        for item in blockers:
            print("-", item["code"], item.get("detail", ""))

    return 0 if status == "PASS_TO_RENDER" else 1

from __future__ import annotations

import argparse

def read_json(path: Path):
    with path.open("r", encoding="utf-8-sig") as f:
        return json.load(f)


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def safe_child(root: Path, relative: str) -> Path:
    root = root.resolve()
    candidate = (root / relative).resolve()
    if candidate != root and root not in candidate.parents:
        raise ValueError(f"Asset path escapes package directory: {relative}")
    return candidate


def validate_reports(script_report_path: Path, visual_report_path: Path):
    errors = []
    warnings = []

    script = read_json(script_report_path)
    visual = read_json(visual_report_path)

    topic_id = script.get("topic_id")
    draft_dir = Path(script.get("draft_directory", "")).resolve()

    if not topic_id:
        errors.append("SCRIPT_TOPIC_ID_MISSING")

    if not draft_dir.is_dir():
        errors.append("DRAFT_DIRECTORY_MISSING")

    if script.get("status") != "SCRIPT_GATES_PASS_VISUAL_RENDER_PENDING":
        errors.append("SCRIPT_PIPELINE_STATUS_NOT_PASS")

    if script.get("fact_check_status") != "PASS":
        errors.append("FACT_CHECK_NOT_PASS")

    if script.get("script_qa_status") != "PASS":
        errors.append("SCRIPT_QA_NOT_PASS")

    if script.get("production_approved") is not False:
        errors.append("PRODUCTION_FLAG_NOT_LOCKED")

    if script.get("render_authorized") is not False:
        errors.append("SCRIPT_RENDER_FLAG_NOT_LOCKED")

    if script.get("upload_authorized") is not False:
        errors.append("SCRIPT_UPLOAD_FLAG_NOT_LOCKED")

    if visual.get("status") != "PASS":
        errors.append("VISUAL_REPORT_NOT_PASS")

    if visual.get("production_approved") is not False:
        errors.append("VISUAL_PRODUCTION_FLAG_NOT_LOCKED")

    if visual.get("render_authorized") is not False:
        errors.append("VISUAL_RENDER_FLAG_NOT_LOCKED")

    if visual.get("upload_authorized") is not False:
        errors.append("VISUAL_UPLOAD_FLAG_NOT_LOCKED")

    manifest_path = Path(visual.get("manifest", "")).resolve()

    if not manifest_path.is_file():
        errors.append("VISUAL_MANIFEST_MISSING")
        return script, visual, None, errors, warnings

    manifest = read_json(manifest_path)
    package_dir = manifest_path.parent

    if manifest.get("topic_id") != topic_id:
        errors.append("TOPIC_ID_MISMATCH")

    manifest_draft = Path(manifest.get("draft_directory", "")).resolve()
    if manifest_draft != draft_dir:
        errors.append("DRAFT_DIRECTORY_MISMATCH")

    if Path(manifest.get("package_directory", "")).resolve() != package_dir:
        errors.append("PACKAGE_DIRECTORY_MISMATCH")

    if visual_report_path.parent.resolve() != package_dir:
        errors.append("VISUAL_REPORT_PACKAGE_MISMATCH")

    if manifest.get("generation_method") != "local_procedural_original":
        warnings.append("UNEXPECTED_GENERATION_METHOD")

    if manifest.get("external_media_used") is not False:
        warnings.append("EXTERNAL_MEDIA_PROVENANCE_REQUIRES_REVIEW")

    assets = manifest.get("assets")
    if not isinstance(assets, list) or not assets:
        errors.append("ASSET_LIST_EMPTY_OR_INVALID")
        return script, visual, manifest, errors, warnings

    expected_total = manifest.get("total_assets")
    if expected_total != len(assets):
        errors.append("MANIFEST_ASSET_COUNT_MISMATCH")

    if visual.get("total") != len(assets):
        errors.append("VISUAL_REPORT_ASSET_COUNT_MISMATCH")

    if visual.get("passed") != len(assets):
        errors.append("VISUAL_REPORT_PASS_COUNT_MISMATCH")

    if visual.get("rejected") != 0:
        errors.append("VISUAL_REPORT_HAS_REJECTED_ASSETS")

    seen_hashes = set()
    seen_ids = set()
    verified = 0
    asset_errors = []

    try:
        from PIL import Image
    except Exception as exc:
        errors.append(f"PILLOW_UNAVAILABLE: {exc}")
        return script, visual, manifest, errors, warnings

    for index, asset in enumerate(assets, start=1):
        asset_id = asset.get("asset_id")
        rel_path = asset.get("asset_path")
        item_errors = []

        if not asset_id or asset_id in seen_ids:
            item_errors.append("MISSING_OR_DUPLICATE_ASSET_ID")
        else:
            seen_ids.add(asset_id)

        if asset.get("status") != "AUTOMATED_TECHNICAL_PASS":
            item_errors.append("ASSET_STATUS_NOT_PASS")

        if not rel_path:
            item_errors.append("ASSET_PATH_MISSING")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        try:
            path = safe_child(package_dir, rel_path)
        except Exception as exc:
            item_errors.append(f"UNSAFE_ASSET_PATH: {exc}")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        if not path.is_file():
            item_errors.append("ASSET_FILE_MISSING")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        actual_hash = sha256_file(path)
        declared_hash = asset.get("sha256")

        if not declared_hash or actual_hash != declared_hash:
            item_errors.append("SHA256_MISMATCH")

        if actual_hash in seen_hashes:
            item_errors.append("EXACT_DUPLICATE_ASSET")
        seen_hashes.add(actual_hash)

        expected_size = (
            (1920, 1080) if asset.get("format") == "landscape"
            else (1080, 1920) if asset.get("format") == "vertical"
            else None
        )

        try:
            with Image.open(path) as im:
                im.verify()

            with Image.open(path) as im:
                actual_size = im.size
                actual_format = im.format

            if expected_size is None:
                item_errors.append("UNKNOWN_ASSET_FORMAT")
            elif actual_size != expected_size:
                item_errors.append(
                    f"WRONG_DIMENSIONS: expected={expected_size}, actual={actual_size}"
                )

            if actual_format != "JPEG":
                item_errors.append(f"WRONG_IMAGE_FORMAT: {actual_format}")

        except Exception as exc:
            item_errors.append(f"IMAGE_DECODE_FAILED: {exc}")

        provenance = asset.get("provenance") or {}
        if provenance.get("source_type") != "local_procedural_original":
            item_errors.append("PROVENANCE_SOURCE_TYPE_INVALID")

        if provenance.get("external_images_used") is not False:
            item_errors.append("EXTERNAL_IMAGE_PROVENANCE_NOT_CLEAR")

        if provenance.get("stock_media_used") is not False:
            item_errors.append("STOCK_MEDIA_PROVENANCE_NOT_CLEAR")

        if not provenance.get("renderer"):
            item_errors.append("RENDERER_PROVENANCE_MISSING")

        if not provenance.get("rights_basis"):
            item_errors.append("RIGHTS_BASIS_MISSING")

        if not asset.get("visual_direction"):
            warnings.append(f"VISUAL_DIRECTION_MISSING:{asset_id}")

        if not asset.get("narration_excerpt") and asset.get("category") != "thumbnail":
            warnings.append(f"NARRATION_EXCERPT_MISSING:{asset_id}")

        if item_errors:
            asset_errors.append({"asset": asset_id, "errors": item_errors})
        else:
            verified += 1

    if len(seen_ids) != len(assets):
        errors.append("ASSET_ID_UNIQUENESS_FAILED")

    if len(seen_hashes) != len(assets):
        errors.append("ASSET_HASH_UNIQUENESS_FAILED")

    if verified != len(assets):
        errors.append("ONE_OR_MORE_ASSETS_FAILED_REVALIDATION")

    if asset_errors:
        errors.append("ASSET_VALIDATION_ERRORS_PRESENT")

    if script.get("public_upload_authorization") is not False:
        errors.append("PUBLIC_UPLOAD_AUTHORIZATION_NOT_LOCKED")

    approval = manifest.get("approval") or {}
    for key in (
        "production_approved",
        "render_authorized",
        "upload_authorized",
        "public_upload_authorized",
    ):
        if approval.get(key) is not False:
            errors.append(f"MANIFEST_APPROVAL_NOT_LOCKED:{key}")

    return script, visual, manifest, errors, warnings, {
        "verified_assets": verified,
        "asset_errors": asset_errors,
        "package_directory": str(package_dir),
        "manifest_path": str(manifest_path),
    }

    def make_csv(self, directory, rows):
        path = Path(directory) / "decisions.csv"
        with path.open("w", encoding="utf-8", newline="") as handle:
            writer = csv.DictWriter(
                handle,
                fieldnames=["video_id", "research_decision", "reviewer_notes", "decided_at"],
            )
            writer.writeheader()
            writer.writerows(rows)
        return path

    def test_pending_is_blocked(self):
        with tempfile.TemporaryDirectory() as directory:
            path = self.make_csv(directory, [{
                "video_id": "pilot", "research_decision": "PENDING",
                "reviewer_notes": "Awaiting review", "decided_at": "",
            }])
            with self.assertRaisesRegex(guard.ResearchApprovalError, "NOT_APPROVED:PENDING"):
                guard.read_research_decision("pilot", path)

    def test_exact_approval_passes(self):
        with tempfile.TemporaryDirectory() as directory:
            path = self.make_csv(directory, [{
                "video_id": "pilot", "research_decision": "APPROVE",
                "reviewer_notes": "Reviewed and approved", "decided_at": "2026-11-18",
            }])
            result = guard.read_research_decision("pilot", path)
            self.assertEqual(result["research_decision"], "APPROVE")

    def test_duplicate_decisions_are_blocked(self):
        with tempfile.TemporaryDirectory() as directory:
            row = {"video_id": "pilot", "research_decision": "APPROVE"}
            path = self.make_csv(directory, [row, row])
            with self.assertRaisesRegex(guard.ResearchApprovalError, "FOUND_2"):
                guard.read_research_decision("pilot", path)

    def test_missing_decision_is_blocked(self):
        with tempfile.TemporaryDirectory() as directory:
            path = self.make_csv(directory, [])
            with self.assertRaisesRegex(guard.ResearchApprovalError, "FOUND_0"):
                guard.read_research_decision("pilot", path)

    def test_pipeline_preflight_blocks_pending(self):
        with tempfile.TemporaryDirectory() as directory:
            path = self.make_csv(directory, [{
                "video_id": "pilot", "research_decision": "PENDING",
            }])
            original = pipeline.DECISIONS_FILE
            pipeline.DECISIONS_FILE = path
            try:
                status, detail = pipeline.canonical_research_decision("pilot")
            finally:
                pipeline.DECISIONS_FILE = original
            self.assertEqual(status, "HOLD")
            self.assertIn("NOT_APPROVED:PENDING", detail)

    def test_generator_blocks_pending_before_api(self):
        with tempfile.TemporaryDirectory() as directory:
            path = self.make_csv(directory, [{
                "video_id": "pilot", "research_decision": "PENDING",
            }])
            original = generator.DECISIONS_FILE
            generator.DECISIONS_FILE = path
            try:
                with self.assertRaisesRegex(RuntimeError, "PENDING"):
                    generator.require_research_approval("pilot")
            finally:
                generator.DECISIONS_FILE = original

    def test_pipeline_requires_explicit_gemini_flag(self):

            pipeline.DECISIONS_FILE = path
            try:
                result = pipeline.run("pilot", execute_gemini=False)
            finally:
                pipeline.DECISIONS_FILE = original
            self.assertEqual(result, 2)
