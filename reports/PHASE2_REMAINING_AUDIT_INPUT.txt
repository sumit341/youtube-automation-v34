ROOT = Path(__file__).resolve().parent.parent
DEFAULT_TOPIC_ID = "E1irX9cDX8U"
SCRIPT_QA_VERSION = "1.3"
DECISIONS_FILE = ROOT / "data" / "research" / "youtube_universe_v33" / "research_review_decisions_v33.csv"
TOPIC_ID_PATTERN = re.compile(r"^[A-Za-z0-9_-]{1,128}$")



def canonical_research_decision(topic_id):
    """Return one canonical human decision; ambiguity and missing rows fail closed."""
    if not DECISIONS_FILE.is_file():
        return "HOLD", "CANONICAL_RESEARCH_DECISION_FILE_MISSING"
    with DECISIONS_FILE.open("r", encoding="utf-8-sig", newline="") as handle:
        rows = list(csv.DictReader(handle))
    matches = [
        row for row in rows
        if str(row.get("video_id", "")).strip() == topic_id
    ]
    if len(matches) != 1:
        return "HOLD", f"EXPECTED_EXACTLY_ONE_DECISION_FOUND_{len(matches)}"
    decision = str(matches[0].get("research_decision", "")).strip().upper()
    if decision not in ("APPROVE", "APPROVE_RESEARCH"):
        return "HOLD", f"RESEARCH_DECISION_NOT_APPROVED:{decision or 'EMPTY'}"
    return "APPROVED", decision


def classify_generation_failure(topic_id, failure_report):
    """Return a fail-closed, structured result for bounded generator exhaustion."""
    if not isinstance(failure_report, dict):
        return None
    if failure_report.get("video_id") != topic_id:
        return None
    if failure_report.get("status") != "NEEDS_REGENERATION":
        return None
    try:
        word_count = int(failure_report.get("long_script_word_count", 0))
        attempts = int(failure_report.get("generation_attempts", 0))
        minimum = int(failure_report.get("minimum_words", 900))
        maximum = int(failure_report.get("maximum_words", 1200))
    except (TypeError, ValueError):
        return None
    if word_count < 0 or attempts < 1 or minimum < 1 or maximum < minimum:
        return None
    return {
        "status": "GENERATION_RETRIES_EXHAUSTED",
        "reason": str(failure_report.get("reason") or "LONG_SCRIPT_COMPLETENESS_RANGE_NOT_MET"),
        "long_script_word_count": word_count,
        "minimum_words": minimum,
        "maximum_words": maximum,
        "generation_attempts": attempts,
        "max_regeneration_attempts": max(0, attempts - 1),
        "production_approved": False,
        "render_authorized": False,
        "upload_authorized": False,
        "public_upload_authorized": False,
    }


def validate_topic_id(topic_id):
    if not isinstance(topic_id, str) or not TOPIC_ID_PATTERN.fullmatch(topic_id):
        raise ValueError("Invalid topic ID. Use 1-128 letters, numbers, underscores, or hyphens.")
    return topic_id


def read_json(path):
    if not path.is_file():
        raise FileNotFoundError(path)
    return json.loads(path.read_text(encoding="utf-8-sig"))


def latest_draft(topic_id):
    base = ROOT / "production" / "runs" / topic_id / "drafts"
    manifests = list(base.glob("*/draft_manifest.json"))
    if not manifests:
        raise FileNotFoundError(f"No draft manifest under {base}")
    return max(manifests, key=lambda p: p.stat().st_mtime).parent.resolve()


def execute_stage(name, command, log_dir):
    log_path = log_dir / f"{name}.log"
    started = datetime.now().astimezone().isoformat()
    with log_path.open("w", encoding="utf-8") as handle:
        result = subprocess.run(
            command,
            cwd=ROOT,
            stdout=handle,
            stderr=subprocess.STDOUT,
            text=True,
            check=False,
        )
    return {
        "stage": name,
        "command": [str(x) for x in command],
        "exit_code": result.returncode,
        "status": "PASS" if result.returncode == 0 else "FAIL",
        "started_at": started,
        "finished_at": datetime.now().astimezone().isoformat(),
        "log": str(log_path.relative_to(ROOT)),
    }


def evaluate_reports(topic_id, draft, fact, qa, completeness):
    problems = []

    if fact.get("report_version") != "2.4":
        problems.append("FACT_REPORT_VERSION_INVALID")
    if fact.get("status") != "PASS":
        problems.append("FACT_CHECK_NOT_PASS")
    if fact.get("topic_id") != topic_id:
        problems.append("FACT_CHECK_TOPIC_MISMATCH")
    if Path(str(fact.get("draft_directory", ""))).resolve() != draft:
        problems.append("FACT_CHECK_DRAFT_MISMATCH")
    if int(fact.get("blocker_count", len(fact.get("blockers", [])))) != 0:
        problems.append("FACT_CHECK_HAS_BLOCKERS")

    if qa.get("qa_version") != SCRIPT_QA_VERSION:
        problems.append("SCRIPT_QA_VERSION_INVALID")
    if qa.get("status") != "PASS":
        problems.append("SCRIPT_QA_NOT_PASS")
    if qa.get("topic_id") != topic_id:
        problems.append("SCRIPT_QA_TOPIC_MISMATCH")
    if Path(str(qa.get("draft_directory", ""))).resolve() != draft:
        problems.append("SCRIPT_QA_DRAFT_MISMATCH")
    if int(qa.get("passed_checks", 0)) != int(qa.get("total_checks", -1)):
        problems.append("SCRIPT_QA_CHECKS_INCOMPLETE")

    if completeness.get("gate_version") != "1.1":
        problems.append("SCRIPT_COMPLETENESS_VERSION_INVALID")
    if completeness.get("video_id") != topic_id:
        problems.append("SCRIPT_COMPLETENESS_TOPIC_MISMATCH")
    if Path(str(completeness.get("draft_directory", ""))).resolve() != draft:
        problems.append("SCRIPT_COMPLETENESS_DRAFT_MISMATCH")
    if completeness.get("status") != "PASS":
        problems.append("SCRIPT_COMPLETENESS_NOT_PASS")
    try:
        long_words = int(completeness.get("word_count", 0))
        long_min = int(completeness.get("minimum_words", 900))
        long_max = int(completeness.get("maximum_words", 1200))
        short_words = int(completeness.get("short_word_count", 0))
        short_min = int(completeness.get("short_minimum_words", 85))
        short_max = int(completeness.get("short_maximum_words", 110))
        if not long_min <= long_words <= long_max:
            problems.append("LONG_SCRIPT_WORD_COUNT_OUT_OF_RANGE")
        if not short_min <= short_words <= short_max:
            problems.append("SHORT_SCRIPT_WORD_COUNT_OUT_OF_RANGE")
    except (TypeError, ValueError):
        problems.append("SCRIPT_COMPLETENESS_WORD_COUNTS_INVALID")

    completeness_safety = completeness.get("safety") or {}
    for key in ("production_approved", "render_authorized", "upload_authorized", "public_upload_authorized"):
        if completeness_safety.get(key) is not False:
            problems.append("COMPLETENESS_SAFETY_LOCK_NOT_FALSE:" + key)

    safety = qa.get("safety") or {}
    for key in ("production_approved", "render_authorized", "upload_authorized"):
        if safety.get(key) is not False:
            problems.append("SAFETY_LOCK_NOT_FALSE:" + key)

    return problems


def self_test():
    test_topic = DEFAULT_TOPIC_ID
    draft = (ROOT / "production" / "runs" / test_topic / "drafts" / "test").resolve()
    fact = {
        "report_version": "2.4", "status": "PASS",
        "topic_id": test_topic, "draft_directory": str(draft),
        "blockers": [],
    }
    completeness = {
        "gate_version": "1.1", "video_id": test_topic,
        "draft_directory": str(draft), "status": "PASS",
        "word_count": 1000, "minimum_words": 900, "maximum_words": 1200,
        "short_word_count": 95, "short_minimum_words": 85, "short_maximum_words": 110,
        "safety": {
            "production_approved": False, "render_authorized": False,
            "upload_authorized": False, "public_upload_authorized": False,
        },
    }
    qa = {
        "qa_version": SCRIPT_QA_VERSION, "status": "PASS",
        "topic_id": test_topic, "draft_directory": str(draft),
        "passed_checks": 23, "total_checks": 23,
        "safety": {
            "production_approved": False,
            "render_authorized": False,
            "upload_authorized": False,
        },
    }
    assert evaluate_reports(test_topic, draft, fact, qa, completeness) == []

    bad_qa = dict(qa)
    bad_qa["safety"] = dict(qa["safety"], upload_authorized=True)
    assert any("SAFETY_LOCK" in x for x in evaluate_reports(test_topic, draft, fact, bad_qa, completeness))

    bad_fact = dict(fact, status="HOLD")
    assert "FACT_CHECK_NOT_PASS" in evaluate_reports(test_topic, draft, bad_fact, qa, completeness)
    bad_completeness = dict(completeness, status="NEEDS_REGENERATION", word_count=295)
    assert "SCRIPT_COMPLETENESS_NOT_PASS" in evaluate_reports(
        test_topic, draft, fact, qa, bad_completeness
    )
    assert "LONG_SCRIPT_WORD_COUNT_OUT_OF_RANGE" in evaluate_reports(
        test_topic, draft, fact, qa, bad_completeness
    )

    bad_qa2 = dict(qa, topic_id="WRONG_TOPIC")
    assert "SCRIPT_QA_TOPIC_MISMATCH" in evaluate_reports(test_topic, draft, fact, bad_qa2, completeness)

    generation_failure = classify_generation_failure(test_topic, {
        "video_id": test_topic,
        "status": "NEEDS_REGENERATION",
        "reason": "LONG_SCRIPT_COMPLETENESS_RANGE_NOT_MET",
        "long_script_word_count": 295,
        "minimum_words": 900,
        "maximum_words": 1200,
        "generation_attempts": 3,
    })
    assert generation_failure is not None
    assert generation_failure["status"] == "GENERATION_RETRIES_EXHAUSTED"
    assert generation_failure["long_script_word_count"] == 295
    assert generation_failure["production_approved"] is False
    assert classify_generation_failure("OTHER_TOPIC", {
        "video_id": test_topic, "status": "NEEDS_REGENERATION",
        "long_script_word_count": 295, "generation_attempts": 3,
    }) is None
    assert classify_generation_failure(test_topic, {
        "video_id": test_topic, "status": "NEEDS_REGENERATION",
        "long_script_word_count": -1, "generation_attempts": 3,
    }) is None

    old_version_qa = dict(qa, qa_version="1.2")
    assert "SCRIPT_QA_VERSION_INVALID" in evaluate_reports(test_topic, draft, fact, old_version_qa, completeness)

    bad_fact_topic = dict(fact, topic_id="DIFFERENT_TOPIC")
    assert "FACT_CHECK_TOPIC_MISMATCH" in evaluate_reports(test_topic, draft, bad_fact_topic, qa, completeness)

    try:
        validate_topic_id("../escape")
    except ValueError:
        pass
    else:
        raise AssertionError("Path traversal topic ID was not rejected")

    print("[PASS] Phase 2 pipeline self-tests: 8 scenarios")
    return 0


def run(topic_id, authorize_render=False, execute_gemini=False, offline_pilot_draft=False, execute_edge_tts=False):
    validate_topic_id(topic_id)
    if execute_gemini and offline_pilot_draft:
        raise ValueError("Choose either Gemini generation or the offline pilot template, not both.")
    if offline_pilot_draft and topic_id != DEFAULT_TOPIC_ID:
        raise ValueError("Offline pilot template is only available for the configured pilot topic.")
    try:
        draft = latest_draft(topic_id)
    except FileNotFoundError:
        draft = ROOT / "production" / "runs" / topic_id / "drafts" / "PENDING_GEMINI"
    stamp = datetime.now().astimezone().strftime("%Y%m%d_%H%M%S_%f")
    report_dir = ROOT / "reports" / "phase2_runs" / stamp
    log_dir = report_dir / "logs"
    log_dir.mkdir(parents=True, exist_ok=False)

    report = {
        "pipeline_version": "1.3.0",
        "topic_id": topic_id,
        "draft_directory": str(draft),
        "started_at": datetime.now().astimezone().isoformat(),
        "stages": [],
        "status": "RUNNING",
        "production_approved": False,
        "render_authorized": False,
        "upload_authorized": False,
        "public_upload_authorization": False,
        "capabilities": {
            "deterministic_fact_traceability": True,
            "script_quality_checks": True,
            "gemini_topic_based_draft_generation": True,
            "automatic_script_regeneration": False,
            "offline_pilot_template_available": True,
            "edge_tts_requires_explicit_authorization": True,
            "bounded_generation_retries_within_authorized_request": True,
            "semantic_truth_verification": False,
            "independent_source_appraisal": False,
        },
        "limitations": [
            "A deterministic PASS does not independently prove scientific truth.",
            "Gemini performs up to three completeness attempts only inside one explicitly authorized API run.",
            "The offline controlled template is pilot-specific and is not a general-purpose generator for new topics.",
            "Microsoft Edge TTS is an external network service and is skipped unless --execute-edge-tts is supplied.",
            "Source authenticity and semantic appraisal require independent verification.",
            "Visual assets, originality and voiceover run after script gates; rendering requires --authorize-render and upload remains disabled.",
        ],
    }

    try:
        decision_status, decision_detail = canonical_research_decision(topic_id)
        report["research_decision_status"] = decision_status
        report["research_decision_detail"] = decision_detail
        if decision_status != "APPROVED":
            report["status"] = "HOLD_RESEARCH_APPROVAL"
            report["failure_queue"] = [{
                "stage": "research_approval_preflight",
                "reason": decision_detail,
                "action": "WAIT_FOR_EXPLICIT_HUMAN_RESEARCH_APPROVAL",
            }]
            report["stages"].append({
                "stage": "research_approval_preflight",
                "status": "BLOCKED",
                "exit_code": None,
                "detail": decision_detail,
            })
            return 2

        if not execute_gemini and not offline_pilot_draft:
            report["status"] = "HOLD_GENERATION_NOT_AUTHORIZED"
            report["failure_queue"] = [{
                "stage": "draft_generation",
                "reason": "NO_GENERATION_MODE_SELECTED",
                "action": "USE_OFFLINE_PILOT_TEMPLATE_OR_EXPLICITLY_AUTHORIZE_GEMINI",
            }]
            report["stages"].append({
                "stage": "draft_generation",
                "status": "SKIPPED_NO_GENERATION_MODE_SELECTED",
                "exit_code": None,
            })
            return 2

        python = sys.executable
        if offline_pilot_draft:
            generator_stage = ("offline_pilot_draft_generation", [
                python, str(ROOT / "production" / "draft_generator_v2.py"),
                "--topic-id", topic_id,
            ])
        else:
            generator_stage = ("gemini_draft_generation", [
                python, str(ROOT / "production" / "gemini_draft_generator_v1.py"),
                "--topic-id", topic_id, "--execute-gemini",
            ])
        stages = [
            generator_stage,
            ("script_completeness_gate", [
                python, str(ROOT / "production" / "script_completeness_gate_v1.py"),
                "--topic-id", topic_id,
            ]),
            ("automated_fact_check", [
                python, str(ROOT / "production" / "automated_fact_check_v2.py"),
                "--topic-id", topic_id,
            ]),
            ("script_qa", [
                python, str(ROOT / "production" / "script_qa_v1.py"),
                "--topic-id", topic_id,
            ]),
        ]

        for name, command in stages:
            stage = execute_stage(name, command, log_dir)
            report["stages"].append(stage)
            if name in ("gemini_draft_generation", "offline_pilot_draft_generation") and stage["status"] == "PASS":
                draft = latest_draft(topic_id)
                report["draft_directory"] = str(draft)
            if stage["status"] != "PASS":
                if name == "script_completeness_gate":
                    completeness_path = draft / "script_completeness_gate_v1_report.json"
                    completeness_report = read_json(completeness_path) if completeness_path.is_file() else {}
                    report["status"] = "HOLD_SCRIPT_COMPLETENESS"
                    report["script_completeness_report"] = str(completeness_path)
                    report["script_completeness_status"] = completeness_report.get("status", "MISSING")
                    report["failure_queue"] = [{
                        "stage": name,
                        "reason": completeness_report.get("regeneration", {}).get("reason", "COMPLETENESS_GATE_FAILED"),
                        "log": stage["log"],
                    }]
                    break

                generation_failure = None
                if name == "gemini_draft_generation":
                    failure_report_path = (
                        ROOT / "production" / "runs" / topic_id
                        / "script_generation_failure_v1_report.json"
                    )
                    if failure_report_path.is_file():
                        try:
                            generation_failure = classify_generation_failure(
                                topic_id, read_json(failure_report_path)
                            )
                        except (OSError, ValueError, TypeError):
                            generation_failure = None

                if generation_failure:
                    report["status"] = "GENERATION_RETRIES_EXHAUSTED"
                    report["generation_failure_report"] = str(failure_report_path)
                    report["generation_failure"] = generation_failure
                    report["failure_queue"] = [{
                        "stage": name,
                        "reason": generation_failure["reason"],
                        "action": "GENERATION_RETRIES_EXHAUSTED_NO_AUTOMATIC_RETRY_SCHEDULED_REVIEW_BEFORE_A_NEW_EXPLICIT_RUN",
                        "long_script_word_count": generation_failure["long_script_word_count"],
                        "required_words": [
                            generation_failure["minimum_words"],
                            generation_failure["maximum_words"],
                        ],
                        "generation_attempts": generation_failure["generation_attempts"],
                        "log": stage["log"],
                    }]
                else:
                    report["status"] = "FAILED_STOPPED"
                    report["failure_queue"] = [{
                        "stage": name,
                        "reason": "PROCESS_EXIT_NONZERO",
                        "log": stage["log"],
                    }]
                break
        else:
            completeness_path = draft / "script_completeness_gate_v1_report.json"
            fact_path = draft / "automated_fact_check_v2_report.json"
            qa_path = draft / "script_qa_report.json"
            completeness = read_json(completeness_path)
            fact = read_json(fact_path)
            qa = read_json(qa_path)
            problems = evaluate_reports(topic_id, draft, fact, qa, completeness)

            report["script_completeness_report"] = str(completeness_path)
            report["script_completeness_status"] = completeness.get("status")
            report["long_script_words"] = completeness.get("word_count")
            report["short_script_words"] = completeness.get("short_word_count")
            report["fact_check_report"] = str(fact_path)
            report["script_qa_report"] = str(qa_path)
            report["fact_check_status"] = fact.get("status")
            report["script_qa_status"] = qa.get("status")
            report["failed_identity_and_safety_checks"] = problems

            if problems:
                report["status"] = "HOLD"
                report["failure_queue"] = [
                    {"reason": problem, "action": "REVIEW_AND_FIX_THEN_RERUN"}
                    for problem in problems
                ]
            else:
                # Visual generation starts only after both script gates pass.
                visual_engine = ROOT / "production" / "automated_visual_engine_v1.py"
                visual_stage = execute_stage(
                    "visual_generation",
                    [python, str(visual_engine), "--topic-id", topic_id],
                    log_dir,
                )
                report["stages"].append(visual_stage)

                if visual_stage["status"] != "PASS":
                    report["status"] = "HOLD_VISUAL_GENERATION_FAILED"
                    report["failure_queue"] = [{
                        "stage": "visual_generation",
                        "reason": "PROCESS_EXIT_NONZERO",
                        "log": visual_stage["log"],
                    }]
                else:
                    packages = sorted(
                        draft.glob("automated_visual_assets_v1_*"),
                        key=lambda item: item.stat().st_mtime,
                        reverse=True,
                    )
                    manifest_path = (
                        packages[0] / "asset_manifest.json"
                        if packages else None
                    )
                    if not manifest_path or not manifest_path.is_file():
                        report["status"] = "HOLD_VISUAL_MANIFEST_MISSING"
                        report["failure_queue"] = [{
                            "stage": "visual_generation",
                            "reason": "ASSET_MANIFEST_NOT_FOUND",
                        }]
                    else:
                        originality_gate = ROOT / "production" / "premium_originality_gate_v1_3.py"
                        originality_stage = execute_stage(
                            "visual_originality_gate",
                            [
                                python, str(originality_gate),
                                "--draft", str(draft),
                                "--manifest", str(manifest_path),
                                "--runs-root", str(ROOT / "production" / "runs"),
                            ],
                            log_dir,
                        )
                        report["stages"].append(originality_stage)
                        gate_report_path = draft / "premium_originality_gate_v1_3_report.json"
                        gate_report = (
                            read_json(gate_report_path)
                            if gate_report_path.is_file() else {}
                        )
                        report["visual_manifest"] = str(manifest_path)
                        report["visual_originality_report"] = str(gate_report_path)
                        report["visual_originality_status"] = gate_report.get("status")
                        report["visual_asset_count"] = gate_report.get("asset_count", 0)

                        if (
                            originality_stage["status"] == "PASS"
                            and gate_report.get("status") == "PASS_ORIGINALITY_GATE"
                            and gate_report.get("topic_id_match") is True
                        ):
                            central_stage = execute_stage(
                                "central_production_gate",
                                [
                                    python,
                                    str(ROOT / "production" / "central_production_gate_v1.py"),
                                    "--topic-id", topic_id,
                                    "--draft", str(draft),
                                ],
                                log_dir,
                            )
                            report["stages"].append(central_stage)
                            central_report_path = draft / "central_production_gate_v1_report.json"
                            central_report = (
                                read_json(central_report_path)
                                if central_report_path.is_file() else {}
                            )
                            report["central_production_gate_report"] = str(central_report_path)
                            report["central_production_gate_status"] = central_report.get("status")

                            if (
                                central_stage["status"] != "PASS"
                                or central_report.get("status") != "PASS_TO_RENDER"
                            ):
                                report["status"] = "HOLD_CENTRAL_PRODUCTION_GATE"
                                report["failure_queue"] = [{
                                    "stage": "central_production_gate",
                                    "reason": "CENTRAL_PRODUCTION_GATE_NOT_PASS",
                                    "gate_status": central_report.get("status"),
                                    "log": central_stage["log"],
                                }]
                            else:
                                report["status"] = "VISUAL_ORIGINALITY_GATES_PASS_RENDER_PENDING"
                                report["failure_queue"] = []
                                report["script_report"] = str(draft / "draft_manifest.json")

                                # Persist the passing state before voiceover reads it.
                                interim_path = report_dir / "phase2_pipeline_report.json"
                                interim_path.write_text(
                                    json.dumps(report, indent=2, ensure_ascii=False),
                                    encoding="utf-8",
                                )
                                if not execute_edge_tts:
                                    report["status"] = "HOLD_VOICEOVER_NOT_AUTHORIZED"
                                    report["voiceover_status"] = "SKIPPED_EXTERNAL_TTS_NOT_AUTHORIZED"
                                    report["stages"].append({
                                        "stage": "voiceover_generation",
                                        "status": "SKIPPED_AWAITING_EXPLICIT_EDGE_TTS_AUTHORIZATION",
                                        "exit_code": None,
                                    })
                                    report["failure_queue"] = [{
                                        "stage": "voiceover_generation",
                                        "reason": "EXPLICIT_EDGE_TTS_AUTHORIZATION_REQUIRED",
                                        "action": "RERUN_WITH_EXECUTE_EDGE_TTS_AFTER_ACCEPTING_EXTERNAL_TTS_SERVICE",
                                    }]
                                else:
                                    voiceover_stage = execute_stage(
                                        "voiceover_generation",
                                        [
                                            python,
                                            str(ROOT / "production" / "automated_voiceover_engine_v1.py"),
                                            "--topic-id", topic_id,
                                            "--execute-edge-tts",
                                        ],
                                        log_dir,
                                    )
                                    report["stages"].append(voiceover_stage)
                                    voiceover_reports = sorted(
                                        draft.glob("voiceover_v1_*/voiceover_manifest.json"),
                                        key=lambda p: p.stat().st_mtime,
                                        reverse=True,
                                    )
                                    voiceover_manifest = (
                                        read_json(voiceover_reports[0])
                                        if voiceover_reports else {}
                                    )
                                    report["voiceover_manifest"] = (
                                        str(voiceover_reports[0]) if voiceover_reports else None
                                    )
                                    report["voiceover_status"] = voiceover_manifest.get("status")
                                    if (
                                        voiceover_stage["status"] == "PASS"
                                        and voiceover_manifest.get("status") == "VOICEOVER_TECHNICAL_PASS"
                                    ):
                                        report["status"] = "VOICEOVER_PASS_RENDER_PENDING"
                                        if authorize_render:
                                            report["render_authorized"] = True
                                            interim_path.write_text(
                                                json.dumps(report, indent=2, ensure_ascii=False),
                                                encoding="utf-8",

def read_json(path):

def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def clean_markdown(text):
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = re.sub(r"```.*?```", " ", text, flags=re.S)
    text = re.sub(r"!\[.*?\]\(.*?\)", " ", text)
    text = re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text)
    text = re.sub(r"^\s{0,3}#{1,6}\s*", "", text, flags=re.M)
    text = re.sub(r"^\s*[-*_]{3,}\s*$", " ", text, flags=re.M)
    text = re.sub(r"^\s*[-*+]\s+", "", text, flags=re.M)
    text = re.sub(r"^\s*\d+[.)]\s+", "", text, flags=re.M)
    text = re.sub(r"^\s*>\s?", "", text, flags=re.M)
    text = re.sub(r"[*_`~]", "", text)
    text = re.sub(r"<[^>]+>", " ", text)
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()



def extract_narration(text):
    """Extract only narration sections; omit metadata, claim maps and draft notes."""
    output = []
    current_heading = None
    body = []

    def flush():
        if current_heading and current_heading.upper() != "DRAFT NOTES":
            output.extend(body)

    for line in text.replace("\r\n", "\n").replace("\r", "\n").splitlines():
        stripped = line.strip()
        if stripped.startswith("## "):
            flush()
            current_heading = stripped[3:].strip()
            body = []
            continue
        if current_heading is None or current_heading.upper() == "DRAFT NOTES":
            continue
        if stripped.lower().startswith("claim mapping:"):
            continue
        body.append(line)

    flush()
    return clean_markdown("\n".join(output))


def word_count(text):
    return len(re.findall(r"\b[\w'-]+\b", text, flags=re.UNICODE))


def get_duration(path):
    ffprobe = shutil.which("ffprobe")
    if not ffprobe:
        raise RuntimeError("ffprobe not found. Install FFmpeg and add it to PATH.")

    result = subprocess.run(
        [
            ffprobe, "-v", "error",
            "-show_entries", "format=duration",
            "-of", "json", str(path)
        ],
        capture_output=True, text=True, check=True
    )
    data = json.loads(result.stdout)
    return float(data["format"]["duration"])


async def generate_audio(text, output_path):

        text=text,
        voice=VOICE,
        rate="+0%",
        volume="+0%",
        pitch="+0Hz"
    )
    await communicate.save(str(output_path))

def read_json(path: Path):

def inside(path: Path, parent: Path) -> bool:
    try:
        path.resolve().relative_to(parent.resolve())
        return True
    except ValueError:
        return False


def probe_duration(audio: Path) -> float:
    ffprobe = shutil.which("ffprobe")
    if not ffprobe:
        raise RuntimeError("ffprobe is required and must be available on PATH.")
    result = subprocess.run(
        [ffprobe, "-v", "error", "-show_entries", "format=duration",
         "-of", "default=noprint_wrappers=1:nokey=1", str(audio)],
        capture_output=True, text=True, check=True,
    )
    duration = float(result.stdout.strip())
    if duration <= 0:
        raise RuntimeError("Audio duration is invalid.")
    return duration


def ffconcat_quote(path: Path) -> str:
    value = str(path.resolve()).replace("\\", "/")
    return "'" + value.replace("'", "'\\''") + "'"


def create_video(topic_id: str, label: str, draft: Path, visual_manifest: Path,
                 voice_manifest: Path, output_dir: Path):
    ffmpeg = shutil.which("ffmpeg")
    if not ffmpeg:
        raise RuntimeError("ffmpeg is required and must be available on PATH.")

    visual = read_json(visual_manifest)
    voice = read_json(voice_manifest)
    package_dir = visual_manifest.parent
    audio_record = (voice.get("audio") or {}).get(label)
    if not audio_record:
        raise RuntimeError(f"{label} voiceover is missing.")

    audio = Path(audio_record.get("file", "")).resolve()
    if not inside(audio, draft) or not audio.is_file():
        raise FileNotFoundError("Voiceover audio is missing or outside the draft directory.")

    category = "long_scene" if label == "long" else "short_scene"
    assets = [
        item for item in visual.get("assets", [])
        if item.get("category") == category
        and item.get("status") == "AUTOMATED_TECHNICAL_PASS"
    ]
    assets.sort(key=lambda item: str(item.get("scene_id", "")))
    if not assets:
        raise RuntimeError(f"No passing {category} visual assets.")

    paths = []
    for item in assets:
        candidate = (package_dir / str(item.get("asset_path", ""))).resolve()
        if not inside(candidate, package_dir) or not candidate.is_file():
            raise RuntimeError("Visual asset path is missing or escapes its package.")
        paths.append(candidate)

    duration = probe_duration(audio)
    per_scene = duration / len(paths)
    concat_file = output_dir / f"{label}_scenes.ffconcat"
    lines = ["ffconcat version 1.0"]
    for path in paths:
        lines.extend([f"file {ffconcat_quote(path)}", f"duration {per_scene:.6f}"])
    lines.append(f"file {ffconcat_quote(paths[-1])}")
    concat_file.write_text("\n".join(lines) + "\n", encoding="utf-8")

    width, height = (1920, 1080) if label == "long" else (1080, 1920)
    output = output_dir / f"{label}_video.mp4"
    command = [
        ffmpeg, "-y", "-hide_banner", "-loglevel", "error",
        "-f", "concat", "-safe", "0", "-i", str(concat_file),
        "-i", str(audio),
        "-vf", f"scale={width}:{height}:force_original_aspect_ratio=increase,crop={width}:{height},fps=30,format=yuv420p",
        "-map", "0:v:0", "-map", "1:a:0",
        "-c:v", "libx264", "-preset", "medium", "-tune", "stillimage",
        "-c:a", "aac", "-b:a", "192k", "-shortest",
        "-movflags", "+faststart", str(output),
    ]
    subprocess.run(command, cwd=ROOT, check=True, capture_output=True, text=True)
    if not output.is_file() or output.stat().st_size < 10000:
        raise RuntimeError(f"{label} MP4 output is missing or unexpectedly small.")
    return {
        "file": str(output),
        "bytes": output.stat().st_size,
        "duration_seconds": round(duration, 2),
        "visual_scenes": len(paths),
        "resolution": f"{width}x{height}",
        "codec": "H.264/AAC",
    }

            lines.append(value)
        return "\n".join(lines)

    for line in text.splitlines():
        if line.startswith("## "):
            if current_heading and current_heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
                sections.extend(current_lines)
            current_heading = line[3:].strip()
            current_lines = []
            continue
        value = line.strip()
        if not current_heading:
            continue
        if value.lower().startswith("claim mapping:") or value.startswith("- "):
            continue
        if value:
            current_lines.append(value)

    if current_heading and current_heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
        sections.extend(current_lines)
    return "\n".join(sections)


def count_narration(text: str) -> int:
    return len(TOKEN_RE.findall(extract_narration(text)))


def safe_script_path(draft_dir: Path, filename: str) -> Path:
    if not isinstance(filename, str) or not filename.strip():
        raise ValueError("Manifest script filename is missing.")
    path = (draft_dir / filename).resolve()
    try:
        path.relative_to(draft_dir.resolve())
    except ValueError as exc:
        raise ValueError("Manifest script path escapes the draft directory.") from exc
    return path


def run_gate(
    topic_id: str,
    minimum: int = 900,
    maximum: int = 1200,
    short_minimum: int = 85,
    short_maximum: int = 110,
) -> int:
    if not isinstance(topic_id, str) or not TOPIC_ID_RE.fullmatch(topic_id):
        raise ValueError("Invalid topic ID.")
    if minimum < 1 or maximum < minimum:
        raise ValueError("Invalid long-form word-count range.")
    if short_minimum < 1 or short_maximum < short_minimum:
        raise ValueError("Invalid Shorts word-count range.")

    run_dir = (ROOT / "production" / "runs" / topic_id).resolve()
    drafts_dir = run_dir / "drafts"
    manifests = list(drafts_dir.glob("*/draft_manifest.json"))
    if not manifests:
        raise FileNotFoundError(f"No draft manifest found for {topic_id}")

    manifest_path = max(manifests, key=lambda p: p.stat().st_mtime)
    draft_dir = manifest_path.parent.resolve()
    manifest = json.loads(manifest_path.read_text(encoding="utf-8-sig"))

    long_path = safe_script_path(draft_dir, (manifest.get("long_script") or {}).get("file"))
    short_path = safe_script_path(draft_dir, (manifest.get("short_script") or {}).get("file"))
    if not long_path.is_file():
        raise FileNotFoundError(f"Long script not found: {long_path}")
    if not short_path.is_file():
        raise FileNotFoundError(f"Short script not found: {short_path}")

    long_words = count_narration(long_path.read_text(encoding="utf-8-sig"))
    short_words = count_narration(short_path.read_text(encoding="utf-8-sig"))
    long_pass = minimum <= long_words <= maximum
    short_pass = short_minimum <= short_words <= short_maximum
    passed = long_pass and short_pass

    failures = []
    if not long_pass:
        failures.append(f"LONG_SCRIPT_WORD_COUNT_{long_words}_OUTSIDE_{minimum}_{maximum}")
    if not short_pass:
        failures.append(f"SHORT_SCRIPT_WORD_COUNT_{short_words}_OUTSIDE_{short_minimum}_{short_maximum}")

    report = {
        "gate_version": "1.1",
        "video_id": topic_id,
        "generated_at": datetime.now(timezone.utc).isoformat(),
        "draft_directory": str(draft_dir),
        "long_script_file": str(long_path),
        "word_count": long_words,
        "minimum_words": minimum,
        "maximum_words": maximum,
        "long_status": "PASS" if long_pass else "NEEDS_REGENERATION",
        "short_script_file": str(short_path),
        "short_word_count": short_words,
        "short_minimum_words": short_minimum,
        "short_maximum_words": short_maximum,
        "short_status": "PASS" if short_pass else "NEEDS_REGENERATION",
        "status": "PASS" if passed else "NEEDS_REGENERATION",
        "regeneration": {
            "required": not passed,
            "status": "NOT_QUEUED" if passed else "QUEUED_NOT_EXECUTED",
            "reason": ";".join(failures),
            "attempts": 0,
            "max_attempts": 2,
        },
        "safety": {
            "original_draft_modified": False,
            "research_decision_modified": False,
            "production_approved": False,
            "render_authorized": False,
            "upload_authorized": False,
            "public_upload_authorized": False,
        },
    }

    report_path = draft_dir / "script_completeness_gate_v1_report.json"
    report_path.write_text(json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8")

    queue_path = run_dir / "script_regeneration_queue_v1.json"
    queue = []
    if queue_path.exists():
        loaded = json.loads(queue_path.read_text(encoding="utf-8-sig"))
        if not isinstance(loaded, list):
            raise ValueError("Regeneration queue must be a JSON list.")
        queue = loaded
    queue = [item for item in queue if item.get("draft_directory") != str(draft_dir)]
    if not passed:
        queue.append({
            "video_id": topic_id,
            "draft_directory": str(draft_dir),
            "reason": report["regeneration"]["reason"],
            "status": "QUEUED_NOT_EXECUTED",
            "attempts": 0,
            "max_attempts": 2,
        })
    queue_path.parent.mkdir(parents=True, exist_ok=True)
    queue_path.write_text(json.dumps(queue, indent=2, ensure_ascii=False), encoding="utf-8")

    print("=" * 58)
    print("SCRIPT COMPLETENESS GATE v1.1")
    print("Video:", topic_id)
    print(f"Long narration: {long_words} words (required {minimum}-{maximum})")
    print(f"Short narration: {short_words} words (required {short_minimum}-{short_maximum})")
    print("Status:", report["status"])
    print("Report:", report_path)
    print("Production/render/upload authorization: False")
    return 0 if passed else 2


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--topic-id", required=True)
    parser.add_argument("--min-words", type=int, default=900)
    parser.add_argument("--max-words", type=int, default=1200)

    ))

from __future__ import annotations

import csv

class ResearchApprovalError(RuntimeError):
    """Raised when a topic lacks one explicit canonical research approval."""


def read_research_decision(topic_id: str, decisions_file: Path = DECISIONS_FILE) -> dict:
    if not decisions_file.is_file():
        raise ResearchApprovalError("CANONICAL_RESEARCH_DECISION_FILE_MISSING")
    with decisions_file.open("r", encoding="utf-8-sig", newline="") as handle:
        reader = csv.DictReader(handle)
        fields = set(reader.fieldnames or [])
        if not {"video_id", "research_decision"}.issubset(fields):
            raise ResearchApprovalError("CANONICAL_RESEARCH_DECISION_SCHEMA_INVALID")
        rows = list(reader)

    matches = [
        row for row in rows
        if str(row.get("video_id", "")).strip() == str(topic_id).strip()
    ]
    if len(matches) != 1:
        raise ResearchApprovalError(
            f"EXPECTED_EXACTLY_ONE_DECISION_FOUND_{len(matches)}"
        )

    row = matches[0]
    decision = str(row.get("research_decision", "")).strip().upper()
    if decision not in ("APPROVE", "APPROVE_RESEARCH"):
        raise ResearchApprovalError(
            f"RESEARCH_DECISION_NOT_APPROVED:{decision or 'EMPTY'}"
        )

    return {
        "video_id": str(topic_id).strip(),
        "research_decision": decision,
        "reviewer_notes": str(row.get("reviewer_notes", "")).strip(),
        "decided_at": str(row.get("decided_at", "")).strip(),
        "source_file": str(decisions_file),
    }


def require_research_approval(topic_id: str) -> dict:
    return read_research_decision(topic_id)

from __future__ import annotations

def load_queue(path: Path):
    if not path.is_file():
        return []
    data = json.loads(path.read_text(encoding="utf-8-sig"))
    if not isinstance(data, list):
        raise ValueError("Regeneration queue must be a JSON list.")
    return data


def save_queue(path: Path, queue):
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(".tmp")
    temp.write_text(json.dumps(queue, indent=2, ensure_ascii=False), encoding="utf-8")
    temp.replace(path)


def main():
    parser = argparse.ArgumentParser(
        description="Bounded, opt-in worker for queued script regeneration."
    )
    parser.add_argument("--topic-id", default="E1irX9cDX8U")
    parser.add_argument("--queue", type=Path, default=None)
    parser.add_argument("--execute", action="store_true",
                        help="Explicitly authorize Gemini regeneration requests.")
    args = parser.parse_args()

    from research_approval_guard_v1 import require_research_approval
    approval = require_research_approval(args.topic_id)
    queue_path = args.queue or (ROOT / "production" / "runs" / args.topic_id / "script_regeneration_queue_v1.json")

    queue = load_queue(queue_path)
    eligible = [
        item for item in queue
        if item.get("video_id") == args.topic_id
        and item.get("status") in ("QUEUED_NOT_EXECUTED", "RETRY_PENDING")
        and int(item.get("attempts", 0)) < int(item.get("max_attempts", 2))
    ][:1]

    print("TOPIC:", args.topic_id)
    print("RESEARCH DECISION:", approval["research_decision"])
    print("ELIGIBLE QUEUE ITEMS:", len(eligible))
    if not args.execute:
        print("MODE: DRY RUN â€” no Gemini request was made.")
        print("Use --execute only when you explicitly authorize API generation.")
        return 0

    if not eligible:
        print("No eligible regeneration items.")
        return 0

    generator = ROOT / "production" / "gemini_draft_generator_v1.py"
    for item in eligible:
        item["attempts"] = int(item.get("attempts", 0)) + 1
        item["last_attempt_at"] = datetime.now(timezone.utc).isoformat()
        item["status"] = "RUNNING"
        save_queue(queue_path, queue)

        result = subprocess.run(
            [sys.executable, str(generator), "--topic-id", args.topic_id, "--execute-gemini"],
            cwd=ROOT,
            check=False,
        )
        item["last_exit_code"] = result.returncode
        item["status"] = "REGENERATED_DRAFT_CREATED" if result.returncode == 0 else (
            "RETRY_PENDING" if item["attempts"] < int(item.get("max_attempts", 2))
            else "EXHAUSTED"
        )
        item["finished_at"] = datetime.now(timezone.utc).isoformat()
        save_queue(queue_path, queue)

        if result.returncode != 0:
            print("Regeneration failed; queue item remains bounded.")
            return result.returncode

    print("Regeneration worker completed.")
    print("Note: regenerated drafts still require fact-check, QA, visual and originality gates.")
    print("Production/render/upload approvals remain false.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

SHORT_REQUIRED = {"HOOK", "EVIDENCE", "QUALIFICATION", "TAKEAWAY"}


def validate_topic_id(topic_id):
    if not isinstance(topic_id, str) or not TOPIC_RE.fullmatch(topic_id):
        raise ValueError("Invalid topic ID.")
    return topic_id


def read_json(path):

def read_csv(path):

def locate_workspace(topic_id):
    for claims_path in (ROOT / "data" / "research").rglob("claims.csv"):
        if claims_path.parent.name == topic_id:
            folder = claims_path.parent
            if all((folder / name).is_file() for name in
                   ("claims.csv", "sources.csv", "evidence.csv")):
                return folder
    raise FileNotFoundError(f"Complete research workspace missing for {topic_id}")



def require_research_approval(topic_id):
    """Fail closed unless the canonical human decision CSV has one explicit approval."""
    if not DECISIONS_FILE.is_file():
        raise RuntimeError("RESEARCH_APPROVAL_BLOCKED: canonical decision CSV is missing.")
    rows = read_csv(DECISIONS_FILE)
    matches = [
        row for row in rows
        if str(row.get("video_id", "")).strip() == topic_id
    ]
    if len(matches) != 1:
        raise RuntimeError(
            "RESEARCH_APPROVAL_BLOCKED: expected exactly one canonical decision "
            f"for {topic_id}; found {len(matches)}."
        )
    decision = str(matches[0].get("research_decision", "")).strip().upper()
    if decision not in ("APPROVE", "APPROVE_RESEARCH"):
        raise RuntimeError(
            f"RESEARCH_APPROVAL_BLOCKED: {topic_id} decision is {decision or 'EMPTY'}."
        )
    return {
        "decision": decision,
        "reviewer_notes": str(matches[0].get("reviewer_notes", "")).strip(),
        "decided_at": str(matches[0].get("decided_at", "")).strip(),
    }


def collect_research(topic_id):
    approval = require_research_approval(topic_id)
    run_dir = ROOT / "production" / "runs" / topic_id
    audit_path = run_dir / "research_consistency_report.json"
    if not audit_path.is_file():
        raise FileNotFoundError(f"Research audit missing: {audit_path}")

    audit = read_json(audit_path)
    if audit.get("topic_id") != topic_id:
        raise RuntimeError("Research audit topic ID mismatch.")
    if audit.get("overall_status") != "READY_FOR_CONTROLLED_DRAFT":
        raise RuntimeError("Research gate is not READY_FOR_CONTROLLED_DRAFT.")

    workspace = locate_workspace(topic_id)
    claims = {x["claim_id"]: x for x in read_csv(workspace / "claims.csv")}
    sources = {x["source_id"]: x for x in read_csv(workspace / "sources.csv")}
    evidence = read_csv(workspace / "evidence.csv")

    eligible = []
    held = []
    references = []

    for result in audit.get("claim_results", []):
        cid = result.get("claim_id")
        if not cid:
            continue
        if result.get("status") == "HOLD":
            held.append(cid)
            continue
        if result.get("status") != "ELIGIBLE_FOR_DRAFT":
            continue
        if cid not in claims:
            raise RuntimeError(f"Eligible claim {cid} missing from claims.csv.")

        linked = []
        evidence_ids = set(result.get("evidence_ids", []))
        for ev in evidence:
            if ev.get("claim_id") != cid or ev.get("evidence_id") not in evidence_ids:
                continue
            src = sources.get(ev.get("source_id"), {})
            if (
                ev.get("evidence_relation", "").upper() == "SUPPORTS"
                and ev.get("review_status", "").upper() == "REVIEWED"
                and src.get("source_review_status", "").upper() == "REVIEWED"
                and ev.get("evidence_summary", "").strip()
                and src.get("title", "").strip()
                and src.get("url_or_doi", "").strip()
            ):
                item = {
                    "evidence_id": ev["evidence_id"],
                    "summary": ev["evidence_summary"],
                    "evidence_notes": ev.get("notes", ""),
                    "source_id": ev["source_id"],
                    "source_title": src["title"],
                    "source_type": src.get("source_type", ""),
                    "source_url": src["url_or_doi"],
                    "source_notes": src.get("notes", ""),
                }
                linked.append(item)
                references.append({"claim_id": cid, **item})

        if not linked:
            raise RuntimeError(f"Eligible claim {cid} has no reviewed supporting evidence.")

        eligible.append({
            "claim_id": cid,
            "claim_text": claims[cid].get("claim_text", ""),
            "claim_type": claims[cid].get("claim_type", ""),
            "limitations": claims[cid].get("limitations", ""),
            "importance": claims[cid].get("importance", ""),
            "evidence": linked,
        })

    if not eligible:
        raise RuntimeError("No eligible, evidence-backed claims found.")

    audit["canonical_research_approval"] = approval
    return audit, eligible, sorted(set(held)), references


def call_gemini(topic_id, eligible, held, revision_feedback=None):
    key = os.getenv("GEMINI_API_KEY")
    if not key:
        raise RuntimeError("GEMINI_API_KEY missing.")

    prompt = f"""
You are a careful US-audience documentary scriptwriter.
Write in natural, clear American English for US viewers interested in
human psychology, money, power and self-improvement.

TOPIC ID: {topic_id}

RESEARCH CLAIMS (the only factual basis you may use):
{json.dumps(eligible, ensure_ascii=False)}

HELD CLAIM IDS â€” NEVER USE:
{json.dumps(held)}

Return ONLY valid JSON with this exact structure:
{{
  "long_video": {{
    "title": "specific documentary title",
    "sections": [
      {{"heading":"HOOK","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"CONTEXT","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"...","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"TAKEAWAY","narration":"...","claim_ids":["C01"],"visual_direction":"..."}}
    ]
  }},
  "short_video": {{
    "title": "...",
    "sections": [
      {{"heading":"HOOK","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"EVIDENCE","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"QUALIFICATION","narration":"...","claim_ids":["C01"],"visual_direction":"..."}},
      {{"heading":"TAKEAWAY","narration":"...","claim_ids":["C01"],"visual_direction":"..."}}
    ]
  }},
  "thumbnail_concepts": [
    {{"concept":"...","composition":"..."}},
    {{"concept":"...","composition":"..."}}
  ],
  "title_ideas": ["...", "...", "..."]
}}

Rules:
- Every section must map only claims that directly support its factual statements.
- Every claim ID must be from the eligible claims list. Never use held claims.
- Do not invent statistics, study details, quotations, people, events, or sources.
- Keep each claim's population, limitations and uncertainty intact.
- Association is not causation. Do not turn observational findings into causal claims.
- Clearly distinguish hypothetical examples from real research.
- For this current pilot, target 900-1200 narration words in the long script.
- Count narration only, excluding headings, claim mappings and draft notes.
- Add useful explanation, evidence context, limitations and concrete examples.
- Never add filler, repetition or invented facts.
- This is a pilot completeness target, not a universal runtime requirement.
- Short-form script must be concise, understandable and self-contained.
- Long sections must include HOOK, CONTEXT and TAKEAWAY.
- Short sections must include HOOK, EVIDENCE, QUALIFICATION and TAKEAWAY.
- Use distinct, specific visual directions for each scene.
- Avoid generic filler, repetition, sensational medical claims and fake certainty.
- Do not include markdown fences or commentary outside the JSON.
"""

    if revision_feedback:
        prompt += (
            "\nREGENERATION FEEDBACK:\n"
            + revision_feedback
            + "\\nRewrite the complete scripts. Do not merely append words.\\n"
        )

    body = {
        "contents": [{"parts": [{"text": prompt}]}],
        "generationConfig": {
            "temperature": 0.45,
            "responseMimeType": "application/json",
            "maxOutputTokens": 8192,
        },
    }

    last_error = None
    for attempt in range(2):
        try:
            response = requests.post(
                API_URL,
                params={"key": key},
                json=body,
                timeout=(15, 120),
            )
            if response.status_code in (429, 400, 401, 403, 404):
                raise RuntimeError(
                    f"GEMINI_API_ERROR_{response.status_code}: "
                    f"{response.text[:700]}"
                )
            if response.status_code in (500, 502, 503, 504):
                last_error = f"GEMINI_TRANSIENT_{response.status_code}"
                if attempt == 0:
                    time.sleep(3)
                    continue
                raise RuntimeError(last_error)

            response.raise_for_status()
            payload = response.json()
            text = payload["candidates"][0]["content"]["parts"][0]["text"]
            result = json.loads(text)
            return result

        except (requests.Timeout, requests.ConnectionError) as exc:
            last_error = f"{type(exc).__name__}: {exc}"
            if attempt == 0:
                time.sleep(3)
                continue
            raise RuntimeError("Gemini connection failed: " + last_error) from exc

    raise RuntimeError("Gemini generation failed: " + str(last_error))


def validate_output(data, eligible, held):
    if not isinstance(data, dict):
        raise ValueError("Gemini output is not a JSON object.")

    allowed = {x["claim_id"] for x in eligible}
    held = set(held)

    for key, required in (
        ("long_video", LONG_REQUIRED),
        ("short_video", SHORT_REQUIRED),
    ):
        obj = data.get(key) or {}
        sections = obj.get("sections")
        if not isinstance(sections, list) or not sections:
            raise ValueError(f"{key} sections missing.")

        headings = set()
        for section in sections:
            heading = str(section.get("heading", "")).strip().upper()
            narration = str(section.get("narration", "")).strip()
            ids = set(section.get("claim_ids") or [])
            if not heading or not narration or not ids:
                raise ValueError(f"Invalid {key} section: {heading}")
            if not ids.issubset(allowed):
                raise ValueError(f"Ineligible claim mapping in {key}/{heading}: {ids-allowed}")
            if ids & held:
                raise ValueError(f"Held claim used in {key}/{heading}.")
            section["heading"] = heading
            headings.add(heading)

        missing = required - headings
        if missing:
            raise ValueError(f"{key} missing required sections: {sorted(missing)}")

    thumbs = data.get("thumbnail_concepts")
    if not isinstance(thumbs, list) or len(thumbs) < 2:
        raise ValueError("At least two thumbnail concepts are required.")

    for title in data.get("title_ideas", []):
        if not isinstance(title, str) or not title.strip():
            raise ValueError("Invalid title idea.")

    return data


def word_count(text):
    return len(re.findall(r"\b[\w'-]+\b", text))

            if all(path.exists() for path in required):
                return claims_path.parent
    raise FileNotFoundError(
        f"Research workspace with claims/evidence/sources not found for {topic_id}"
    )


def latest_draft(run_dir):
    manifests = list((run_dir / "drafts").glob("*/draft_manifest.json"))
    if not manifests:
        raise FileNotFoundError(f"No draft manifest found under {run_dir / 'drafts'}")
    return max(manifests, key=lambda path: path.stat().st_mtime)


def add(checks, name, passed, detail, severity="BLOCKER"):
    checks.append({
        "name": name,
        "passed": bool(passed),
        "severity": severity,
        "detail": str(detail),
    })


def parse_long(text):
    sections = {}
    mappings = {}
    heading = None
    body = []

    def save():
        if heading and heading.upper() != "DRAFT NOTES":
            sections[heading] = "\n".join(body).strip()

    for line in text.splitlines():
        if line.startswith("## "):
            save()
            heading = line[3:].strip()
            body = []
        elif heading:
            if line.strip().lower().startswith("claim mapping:"):
                mappings[heading] = re.findall(r"\bC\d{2}\b", line.upper())
            elif heading.upper() != "DRAFT NOTES":
                body.append(line)

    save()
    return sections, mappings


def parse_short(text, manifest):
    narration = []
    for line in text.splitlines():
        stripped = line.strip()
        if not stripped or stripped.startswith("#") or stripped.startswith("-"):
            continue
        if stripped.lower().startswith("claim mapping:"):
            continue
        narration.append(stripped)
    claim_ids = set(manifest.get("short_script", {}).get("claim_ids", []))
    return "\n".join(narration), claim_ids


def normalize_numbers(text):
    return {
        token.replace(",", "")
        for token in re.findall(r"-?\d[\d,]*(?:\.\d+)?", text or "")
    }


def numeric_fact_phrases(text):
    patterns = [
        r"\b\d[\d,]*(?:\.\d+)?\s*(?:participants?|undergraduates?|students?|studies|trials|"
        r"randomized controlled trials|effect sizes|adults?|publications?|models?|"
        r"cognitive factors)\b",
        r"\bN\s*=\s*\d[\d,]*(?:\.\d+)?",
        r"\bHedges\s*g\s*(?:of|=|~|approximately)?\s*-?\d+(?:\.\d+)?",
        r"\bg\s*=\s*-?\d+(?:\.\d+)?",
    ]
    found = []
    for pattern in patterns:
        found.extend(re.findall(pattern, text or "", flags=re.I))
    return found


def causal_patterns():
    return [
        r"\bcauses?\s+(?:insomnia|sleep impairment|sleep problems?|overthinking)\b",
        r"\bleads?\s+to\s+(?:insomnia|sleep impairment|sleep problems?)\b",
        r"\bresults?\s+in\s+(?:insomnia|sleep impairment|sleep problems?)\b",
        r"\b(?:overthinking|worry|rumination)\s+(?:directly\s+)?causes?\b",
        r"\bproves?\s+that\b",
        r"\b(?:insomnia|sleep problems?)\s+ki wajah se overthinking\b",
        r"\boverthinking\s+ki wajah se\s+(?:insomnia|sleep problems?)\b",
        r"\bisse\s+insomnia\s+hoti\s+hai\b",
    ]


def main():
    parser = argparse.ArgumentParser(
        description="Deterministic claim/evidence/script consistency gate (v2.4)."
    )
    parser.add_argument("--topic-id", default="E1irX9cDX8U")
    args = parser.parse_args()

    topic_id = args.topic_id
    from research_approval_guard_v1 import require_research_approval
    require_research_approval(topic_id)
    run_dir = ROOT / "production" / "runs" / topic_id
    audit_path = run_dir / "research_consistency_report.json"
    if not audit_path.exists():
        raise FileNotFoundError(audit_path)

    workspace = locate_workspace(topic_id)
    manifest_path = latest_draft(run_dir)
    draft_dir = manifest_path.parent
    manifest = read_json(manifest_path)
    audit = read_json(audit_path)

    long_path = draft_dir / manifest["long_script"]["file"]
    short_path = draft_dir / manifest["short_script"]["file"]
    long_text = long_path.read_text(encoding="utf-8-sig")
    short_text = short_path.read_text(encoding="utf-8-sig")

    sections, mappings = parse_long(long_text)
    short_narration, short_ids = parse_short(short_text, manifest)

    claims = {row["claim_id"]: row for row in read_csv(workspace / "claims.csv")}
    evidence_rows = read_csv(workspace / "evidence.csv")
    sources = {row["source_id"]: row for row in read_csv(workspace / "sources.csv")}
    audit_claims = {
        row["claim_id"]: row
        for row in audit.get("claim_results", [])
        if row.get("claim_id")
    }

    checks = []
    used_ids = set(short_ids)
    mapped_passages = {}

    add(
        checks, "RESEARCH_GATE",
        audit.get("overall_status") == "READY_FOR_CONTROLLED_DRAFT",
        audit.get("overall_status", "MISSING"),
    )
    add(
        checks, "TOPIC_ID_MATCH",
        manifest.get("topic_id") == topic_id == audit.get("topic_id"),
        f"Requested={topic_id}; manifest={manifest.get('topic_id')}; audit={audit.get('topic_id')}",
    )

    # Phase 3: Strict Held-Claim Quarantine check
    held_claims = {
        cid for cid, ac in audit_claims.items()
        if ac.get("status") == "HOLD"
    }

    narrative_headings = [
        heading for heading in sections
        if heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}
    ]
    add(
        checks, "LONG_SECTION_MAPPING_COVERAGE",
        bool(narrative_headings)
        and all(heading in mappings and mappings[heading] for heading in narrative_headings),
        f"Mapped sections={sum(bool(mappings.get(h)) for h in narrative_headings)}/{len(narrative_headings)}",
    )

    for heading, ids in mappings.items():
        passage = sections.get(heading, "")

        # Check that no held claim is in this mapping
        mapped_held = set(ids) & held_claims
        add(
            checks, f"SECTION_{heading}_NO_HELD_CLAIMS",
            not bool(mapped_held),
            f"Held claims present in section mapping: {sorted(mapped_held)}",
            "BLOCKER"
        )

        add(
            checks, f"SECTION_{heading}_HAS_TEXT_AND_CLAIMS",
            bool(passage.strip()) and bool(ids),
            f"Text present={bool(passage.strip())}; mapped claims={ids}",
        )
        for claim_id in ids:
            used_ids.add(claim_id)
            mapped_passages.setdefault(claim_id, []).append(
                {
                    "format": "long",
                    "section": heading,
                    "text": passage,
                    "claim_ids": list(ids),
                }
            )

    short_held = short_ids & held_claims
    add(
        checks, "SHORT_SCRIPT_NO_HELD_CLAIMS",
        not bool(short_held),
        f"Held claims present in short script: {sorted(short_held)}",
        "BLOCKER"
    )

    add(
        checks, "SHORT_SCRIPT_MAPPING",
        bool(short_ids) and bool(short_narration.strip()),
        f"Claim IDs={sorted(short_ids)}; narration present={bool(short_narration.strip())}",
    )
    for claim_id in short_ids:
        mapped_passages.setdefault(claim_id, []).append(
            {
                "format": "short",
                "section": "Short narration",
                "text": short_narration,
                "claim_ids": sorted(short_ids),
            }
        )

    for claim_id in sorted(used_ids):
        claim = claims.get(claim_id)
        audited = audit_claims.get(claim_id)

        add(
            checks, f"{claim_id}_EXISTS",
            claim is not None and audited is not None,
            "Claim found in CSV and research audit." if claim and audited
            else "Claim missing from CSV or research audit.",
        )
        if not claim or not audited:
            continue

        add(
            checks, f"{claim_id}_ELIGIBLE",
            audited.get("status") == "ELIGIBLE_FOR_DRAFT",
            f"Audit status={audited.get('status')}",
        )

        limitation = (claim.get("limitations") or "").strip()
        audit_limitation = (audited.get("limitations") or "").strip()
        add(
            checks, f"{claim_id}_LIMITATIONS_RECORDED",
            bool(limitation and audit_limitation),
            f"CSV limitation present={bool(limitation)}; audit limitation present={bool(audit_limitation)}",
        )

        evidence_ids = set(audited.get("evidence_ids", []))
        linked = [
            row for row in evidence_rows
            if row.get("evidence_id") in evidence_ids
            and row.get("claim_id") == claim_id
        ]
        add(
            checks, f"{claim_id}_EVIDENCE_LINK",
            bool(evidence_ids) and bool(linked),
            f"Audit IDs={sorted(evidence_ids)}; matching evidence rows={len(linked)}",
        )

        valid_context = []
        independent_sources = set()
        for ev in linked:
            source = sources.get(ev.get("source_id"), {})
            if (
                ev.get("evidence_relation", "").upper() == "SUPPORTS"
                and ev.get("review_status", "").upper() == "REVIEWED"
                and source.get("source_review_status", "").upper() == "REVIEWED"
                and ev.get("evidence_summary", "").strip()
                and source.get("title", "").strip()
                and source.get("url_or_doi", "").strip()
            ):
                independent_sources.add(ev.get("source_id"))
                valid_context.append(
                    " ".join([
                        ev.get("evidence_summary", ""),
                        ev.get("notes", ""),
                        source.get("title", ""),
                        source.get("notes", ""),
                    ])
                )

        add(
            checks, f"{claim_id}_REVIEWED_SOURCE_EVIDENCE",
            bool(valid_context),
            f"Valid reviewed evidence records={len(valid_context)}",
        )

        # Phase 3: Enhanced independent source verification
        add(
            checks, f"{claim_id}_INDEPENDENT_SOURCE_APPRAISAL",
            len(independent_sources) >= 1,
            f"Verified independent sources backing claim: {sorted(independent_sources)}",
        )

        passages = mapped_passages.get(claim_id, [])
        add(
            checks, f"{claim_id}_HAS_MAPPED_PASSAGE",
            bool(passages),
            f"Mapped passages={len(passages)}",
        )
        if not passages or not valid_context:
            continue

        claim_type = (claim.get("claim_type") or "").upper()
        passage_text = " ".join(item["text"] for item in passages)
        passage_lower = passage_text.lower()

        def context_for_passage(item):
            mapped_ids = set(item.get("claim_ids") or [claim_id])
            mapped_ids.add(claim_id)
            context_parts = []

            for mapped_id in sorted(mapped_ids):
                mapped_claim = claims.get(mapped_id, {})
                mapped_audit = audit_claims.get(mapped_id, {})
                context_parts.append(mapped_claim.get("claim_text", ""))

                mapped_evidence_ids = set(
                    mapped_audit.get("evidence_ids", [])
                )

                for mapped_ev in evidence_rows:
                    if (
                        mapped_ev.get("evidence_id") not in mapped_evidence_ids
                        or mapped_ev.get("claim_id") != mapped_id
                    ):
                        continue

                    mapped_source = sources.get(
                        mapped_ev.get("source_id"), {}
                    )

                    if (
                        mapped_ev.get("evidence_relation", "").upper() == "SUPPORTS"
                        and mapped_ev.get("review_status", "").upper() == "REVIEWED"
                        and mapped_source.get("source_review_status", "").upper() == "REVIEWED"
                        and mapped_ev.get("evidence_summary", "").strip()
                        and mapped_source.get("title", "").strip()
                        and mapped_source.get("url_or_doi", "").strip()
                    ):
                        context_parts.extend([
                            mapped_ev.get("evidence_summary", ""),
                            mapped_ev.get("notes", ""),
                            mapped_source.get("title", ""),
                            mapped_source.get("notes", ""),
                        ])

            return " ".join(context_parts)

        if claim_type == "EMPIRICAL_ASSOCIATION":
            risky = [
                pattern for pattern in causal_patterns()
                if re.search(pattern, passage_lower, flags=re.I)
            ]
            add(
                checks, f"{claim_id}_CAUSAL_OVERSTATEMENT",
                not risky,
                f"Potentially risky patterns={risky}",
            )

        unsupported_numeric = []
        unsupported_populations = []

        population_terms = [
            "undergraduate", "undergraduates", "students", "children",
            "adolescents", "older adults", "adults", "patients",
            "people with insomnia", "people with sleep problems",
        ]

        population_aliases = {
            "students": ("student", "students", "undergraduate", "undergraduates"),
            "undergraduate": ("undergraduate", "undergraduates", "student", "students"),
            "undergraduates": ("undergraduate", "undergraduates", "student", "students"),
        }

        for passage_item in passages:
            item_text = passage_item.get("text", "")
            item_lower = item_text.lower()
            item_context = context_for_passage(passage_item)
            item_context_lower = item_context.lower()
            section_name = passage_item.get("section", "passage")

            for phrase in numeric_fact_phrases(item_text):
                phrase_numbers = normalize_numbers(phrase)
                context_numbers = normalize_numbers(item_context)

                if not phrase_numbers.issubset(context_numbers):
                    unsupported_numeric.append(
                        f"{section_name}: {phrase}"
                    )

            item_populations = [
                term for term in population_terms if term in item_lower
            ]

            for term in item_populations:
                aliases = population_aliases.get(term, (term,))
                if not any(alias in item_context_lower for alias in aliases):
                    unsupported_populations.append(
                        f"{section_name}: {term}"
                    )

        add(
            checks, f"{claim_id}_NUMERIC_FACT_TRACEABILITY",
            not unsupported_numeric,
            f"Unmatched study/statistical quantities={unsupported_numeric}",
            "WARNING",
        )

        add(
            checks, f"{claim_id}_POPULATION_TRACEABILITY",
            not unsupported_populations,
            f"Population terms absent from passage-mapped evidence={unsupported_populations}",
            "WARNING",
        )

        if claim_type == "EMPIRICAL_ASSOCIATION":
            qualification_terms = [
                "association", "associated", "not causal", "does not prove",
                "doesn't prove", "not prove", "not mean", "not universal",
                "cannot conclude", "cannot be concluded",
            ]
            has_qualification = any(term in passage_lower for term in qualification_terms)
            add(
                checks, f"{claim_id}_ASSOCIATION_QUALIFICATION",
                has_qualification,
                "Association/correlation qualification detected in mapped passage."
                if has_qualification else "No configured association qualification phrase detected.",
                "WARNING",
            )

    blockers = [
        item for item in checks
        if not item["passed"] and item["severity"] == "BLOCKER"
    ]
    warnings = [
        item for item in checks
        if not item["passed"] and item["severity"] == "WARNING"
    ]
    status = "FAIL" if blockers else "HOLD" if warnings else "PASS"

    report = {
        "report_version": "2.4",
        "checker": "deterministic_claim_evidence_consistency",
        "created_at": datetime.now().astimezone().isoformat(),
        "topic_id": topic_id,
        "draft_directory": str(draft_dir),
        "status": status,
        "claims_checked": len(used_ids),
        "passed_checks": sum(item["passed"] for item in checks),
        "total_checks": len(checks),
        "blockers": blockers,
        "warnings": warnings,
        "checks": checks,
        "capabilities": {
            "claim_evidence_traceability": True,
            "source_review_metadata_check": True,
            "independent_source_appraisal": False,
            "semantic_truth_verification": False,
            "live_source_fetch": False,
        },
        "limitations": [
            "This is a deterministic consistency and traceability gate (v2.4) with strict held-claim quarantine.",
            "It checks source review-status metadata and evidence linkage; it does not independently fetch or authenticate sources.",
            "Numeric and population checks use configured text matching and may require review when phrasing differs.",
            "A PASS is not proof of scientific truth.",
            "No production, rendering, or upload authorization is granted.",
        ],
        "safety": {
            "fact_review_approved": False,
            "production_approved": False,
            "render_authorized": False,
            "upload_authorized": False,
        },
    }

    output = draft_dir / "automated_fact_check_v2_report.json"
    output.write_text(json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8")

    print("AUTOMATED FACT CHECK v2.4")
    print("=" * 60)
    print("Topic:", topic_id)
    print("Draft:", draft_dir)
    print("Status:", status)
    print("Claims checked:", len(used_ids))
    print("Checks:", report["passed_checks"], "/", report["total_checks"])
    print("Blockers:", len(blockers))
    print("Warnings:", len(warnings))
    for item in checks:
        label = "PASS" if item["passed"] else item["severity"]
        print(f"{label:8} | {item['name']} | {item['detail']}")
    print("Report:", output)
    print("Production approval: NOT GRANTED")
    print("Upload authorization: NOT GRANTED")

from pathlib import Path
from datetime import datetime

ROOT = Path(__file__).resolve().parent.parent

TOPIC_ID_PATTERN = re.compile(r"^[A-Za-z0-9_-]{1,128}$")

def validate_topic_id(topic_id):
    if not isinstance(topic_id, str) or not TOPIC_ID_PATTERN.fullmatch(topic_id):
        raise ValueError("Invalid topic ID. Use 1-128 letters, numbers, underscores, or hyphens.")
    return topic_id

TARGETS = {
    "long": {"minimum": 900, "maximum": 1200, "wpm": 130},
    "short": {"minimum": 85, "maximum": 110, "wpm": 130},
}


def read_json(path):
    if not path.exists():
        raise FileNotFoundError(path)
    return json.loads(path.read_text(encoding="utf-8"))


def read_text(path):
    if not path.exists():
        raise FileNotFoundError(path)
    return path.read_text(encoding="utf-8")


def word_count(text):
    return len(re.findall(r"\b[\w'-]+\b", text))


def extract_mapping_ids(text):
    found = set()

    for line in text.splitlines():
        if line.strip().lower().startswith("claim mapping:"):
            found.update(re.findall(r"\bC\d{2}\b", line))

    return found


def extract_narration(text):
    sections = []
    current_heading = ""
    current_lines = []

    for line in text.splitlines():
        if line.startswith("## "):
            if current_heading and current_lines:
                if current_heading.upper() not in {
                    "DRAFT NOTES", "CLAIM MAPPING"
                }:
                    sections.extend(current_lines)

            current_heading = line[3:].strip()
            current_lines = []
            continue

        if line.strip().startswith("Claim mapping:"):
            continue

        if current_heading:
            current_lines.append(line)

    if current_heading and current_heading.upper() not in {
        "DRAFT NOTES", "CLAIM MAPPING"
    }:
        sections.extend(current_lines)

    return "\n".join(sections)


def add_check(checks, name, passed, detail, severity="BLOCKER"):
    checks.append({
        "name": name,
        "passed": bool(passed),
        "severity": severity,
        "detail": detail,
    })


def check_sections(text, required):
    headings = {
        x.strip().upper()
        for x in re.findall(r"^##\s+(.+?)\s*$", text, re.MULTILINE)
    }
    return {x: x in headings for x in required}


def main():
    parser = argparse.ArgumentParser(description="Automated script QA for a research topic")
    parser.add_argument("--topic-id", default="E1irX9cDX8U")

    audit_file = run_dir / "research_consistency_report.json"

    draft_manifests = sorted(
        (run_dir / "drafts").glob("*/draft_manifest.json"),
        key=lambda p: p.stat().st_mtime,
        reverse=True,
    )

    if not draft_manifests:
        raise FileNotFoundError("No draft manifest found.")

    manifest_path = draft_manifests[0]
    draft_dir = manifest_path.parent
    manifest = read_json(manifest_path)
    research = read_json(audit_file)

    long_text = read_text(draft_dir / manifest["long_script"]["file"])
    short_text = read_text(draft_dir / manifest["short_script"]["file"])

    eligible = {
        x["claim_id"] for x in research["claim_results"]
        if x["status"] == "ELIGIBLE_FOR_DRAFT"
    }
    held = {
        x["claim_id"] for x in research["claim_results"]
        if x["status"] == "HOLD"
    }

    checks = []

    add_check(
        checks, "RESEARCH_GATE",
        research.get("overall_status") == "READY_FOR_CONTROLLED_DRAFT",
        research.get("overall_status", "MISSING"),
    )

    for label, text, target, required in [
        ("LONG", long_text, TARGETS["long"],
         ["HOOK", "CONTEXT", "TAKEAWAY"]),
        ("SHORT", short_text, TARGETS["short"],
         ["HOOK", "EVIDENCE", "QUALIFICATION", "TAKEAWAY"]),
    ]:
        narration = extract_narration(text)
        words = word_count(narration)
        mapping = extract_mapping_ids(text)

        in_target = target["minimum"] <= words <= target["maximum"]
        add_check(

            checks, f"{label}_CLAIM_MAPPING",

            f"Mapped: {sorted(mapping)}; eligible: {sorted(eligible)}",
        )

        add_check(
            checks, f"{label}_HELD_CLAIM_MAPPING",
            not bool(mapping & held),
            f"Held claims in explicit mapping: {sorted(mapping & held)}",

        # Detect literal claim IDs in narration separately.

            re.findall(r"\bC\d{2}\b", narration)
        )

        add_check(

            f"Held claim IDs mentioned in narration: {sorted(narration_mentions & held)}",

        fact_report_path.exists()
        and fact_report_path.stat().st_mtime >= manifest_path.stat().st_mtime
    )

    add_check(
        checks, "FACT_CHECK_REPORT_VERSION",
        report_version_ok,
        f"Expected 2.4; found {fact_report.get('report_version', 'MISSING')}",
    )

    add_check(
        checks, "FACT_CHECK_TOPIC_MATCH",
        report_topic_ok,
        f"Expected {args.topic_id}; found {fact_report.get('topic_id', 'MISSING')}",
    )

    add_check(
        checks, "FACT_CHECK_DRAFT_MATCH",
        report_draft_ok,
        f"Report draft directory matches current draft: {report_draft_ok}",
    )

    add_check(
        checks, "FACT_CHECK_REPORT_FRESHNESS",
        report_fresh_ok,
        f"Report exists and is not older than draft manifest: {report_fresh_ok}",
    )

    add_check(
        checks, "AUTOMATED_FACT_CHECK",
        report_version_ok
        and report_topic_ok
        and report_draft_ok

def load_json(path: Path) -> dict:
    if not path.is_file():
        raise FileNotFoundError(str(path))
    data = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(data, dict):
        raise ValueError(f"Expected JSON object: {path}")
    return data


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def inside(child: Path, parent: Path) -> bool:
    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except ValueError:
        return False


def add(items: list, code: str, detail: str = ""):
    items.append({"code": code, "detail": detail})


def count_narration_words(text: str) -> int:
    sections = []
    heading = None
    lines = []
    for line in text.splitlines():
        if line.startswith("## "):
            if heading and heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
                sections.extend(lines)
            heading = line[3:].strip()
            lines = []
            continue
        value = line.strip()
        if not heading:
            continue
        if value.lower().startswith("claim mapping:") or value.startswith("- "):
            continue
        if value:
            lines.append(value)
    if heading and heading.upper() not in {"DRAFT NOTES", "CLAIM MAPPING"}:
        sections.extend(lines)
    return len(re.findall(r"\b[\wâ€™'-]+\b", " ".join(sections)))

def normalize_text(value):
    value = re.sub(r"```.*?```", " ", value, flags=re.S)
    value = re.sub(r"^\s*#{1,6}\s+.*$", " ", value, flags=re.M)
    value = re.sub(r"https?://\S+", " ", value)
    value = value.lower()
    value = re.sub(r"[^a-z0-9\u0900-\u097f]+", " ", value)

    return " ".join(
        word for word in value.split()
        if len(word) > 2 and word not in IGNORED_WORDS
    )


def make_shingles(text, size=5):
    words = text.split()
    if len(words) < size:
        return {" ".join(words)} if words else set()

    return {
        " ".join(words[index:index + size])
        for index in range(len(words) - size + 1)
    }


def jaccard_similarity(first, second):
    if not first or not second:
        return 0.0

    union = first | second
    return len(first & second) / len(union) if union else 0.0


def average_hash(path):
    with Image.open(path) as image:
        image = image.convert("L").resize((16, 16))

        try:
            pixels = list(image.get_flattened_data())
        except AttributeError:
            pixels = list(image.getdata())

    average = sum(pixels) / len(pixels)
    value = 0

    for pixel in pixels:
        value = (value << 1) | int(pixel >= average)

    return value


def hash_similarity(first, second):
    return 1.0 - ((first ^ second).bit_count() / 256)

def load_script(path):
    if not path.is_file():
        return ""

    return normalize_text(
        path.read_text(encoding="utf-8", errors="replace")
    )


def collect_other_topic_scripts(runs_root, current_draft, kind):
    filename = SCRIPT_NAMES[kind]
    current_topic = topic_id_from_draft(current_draft)
    results = []

    if not runs_root.is_dir():
        return results

    for path in runs_root.rglob(filename):
        try:
            resolved = path.resolve()

            if current_draft.resolve() in resolved.parents:
                continue

            old_topic = topic_id_from_script(resolved)

            if not old_topic or old_topic == current_topic:
                continue

            content = load_script(resolved)

            if content:
                results.append({
                    "path": resolved,
                    "topic_id": old_topic,
                    "content": content,
                })

        except (OSError, RuntimeError):
            continue

    return results


def classify_script_corpus(previous_scripts, comparisons, jaccard_threshold, sequence_threshold):
    """Classify project-local script comparison; first script seeds the corpus."""
    if not previous_scripts:
        return "BASELINE_SEED_NO_PRIOR_TOPIC_SCRIPTS"
    if any(
        item["jaccard_similarity"] >= jaccard_threshold
        or item["sequence_similarity"] >= sequence_threshold
        for item in comparisons
    ):
        return "HOLD_POSSIBLE_REUSE"
    return "COMPARISON_CLEAR"


def validate_provenance(asset):
    provenance = asset.get("provenance") or {}

    failures = []
    holds = []
    source_type = str(
        provenance.get("source_type", "")
    ).strip().lower()

    if source_type == "original":
        if not provenance.get("creator"):
            holds.append("ORIGINAL_CREATOR_MISSING")

        if provenance.get("rights_confirmation") is not True:
            holds.append("ORIGINAL_RIGHTS_NOT_CONFIRMED")

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("ORIGINAL_COMMERCIAL_USE_NOT_CONFIRMED")

        if not (
            provenance.get("source_url")
            or provenance.get("source_code_reference")
            or provenance.get("generation_prompt_used")
        ):
            holds.append("ORIGINAL_SOURCE_TRACEABILITY_MISSING")

    elif source_type == "ai_generated":
        required = (
            "provider",
            "model",
            "terms_reference",
            "terms_checked_at",
            "generation_prompt_used",
        )

        missing = [
            field for field in required
            if not provenance.get(field)
        ]

        if missing:
            holds.append(
                "AI_PROVENANCE_MISSING_" + "_".join(missing)
            )

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("AI_COMMERCIAL_USE_NOT_CONFIRMED")

        if provenance.get("rights_confirmation") is not True:
            holds.append("AI_RIGHTS_NOT_CONFIRMED")

    elif source_type == "local_procedural_original":
        required = ("creator", "source_code_reference", "rights_basis")
        missing = [field for field in required if not provenance.get(field)]
        if missing:
            holds.append("LOCAL_PROCEDURAL_PROVENANCE_MISSING_" + "_".join(missing))
        if provenance.get("external_images_used") is not False:
            holds.append("LOCAL_PROCEDURAL_EXTERNAL_INPUTS_NOT_EXCLUDED")
        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("LOCAL_PROCEDURAL_COMMERCIAL_USE_NOT_CONFIRMED")

    elif source_type == "stock":
        if not provenance.get("source_url"):
            holds.append("STOCK_SOURCE_URL_MISSING")

        if not provenance.get("license_name"):
            holds.append("STOCK_LICENSE_MISSING")

        if not provenance.get("license_url"):
            holds.append("STOCK_LICENSE_REFERENCE_MISSING")

        if provenance.get("commercial_use_confirmed") is not True:
            holds.append("STOCK_COMMERCIAL_USE_NOT_CONFIRMED")

    else:
        holds.append("UNKNOWN_OR_UNDECLARED_SOURCE_TYPE")

    return failures, holds


def validate_asset_path(manifest_dir, relative_path):
    if not relative_path:
        return None, "ASSET_PATH_MISSING"

    candidate = (manifest_dir / relative_path).resolve()

    try:
        candidate.relative_to(manifest_dir.resolve())
    except ValueError:
        return None, "ASSET_PATH_ESCAPES_PACKAGE"

    if not candidate.is_file():
        return None, "ASSET_FILE_MISSING"

    return candidate, None


def main():
    parser = argparse.ArgumentParser(
        description="Read-only Premium Originality Gate V1.2"
    )

    parser.add_argument("--draft", required=True)
    parser.add_argument("--manifest", required=True)
    parser.add_argument(
        "--runs-root",
        default="production/runs",
    )
    parser.add_argument(
        "--visual-warning-threshold",
        type=float,
        default=0.94,
    )
    parser.add_argument(
        "--script-jaccard-threshold",
        type=float,
        default=0.30,
    )
    parser.add_argument(
        "--script-sequence-threshold",
        type=float,
        default=0.65,
    )

    args = parser.parse_args()

    draft = Path(args.draft).resolve()
    manifest_path = Path(args.manifest).resolve()
    runs_root = Path(args.runs_root).resolve()

    if not draft.is_dir():
        print("ERROR: Draft directory not found:", draft)
        return 2

    if not manifest_path.is_file():
        print("ERROR: Manifest not found:", manifest_path)
        return 2

    try:
        manifest = json.loads(
            manifest_path.read_text(encoding="utf-8")
        )
    except Exception as exc:
        print("ERROR: Manifest read failed:", exc)
        return 2

    assets = manifest.get("assets")

    if not isinstance(assets, list):
        print("ERROR: Manifest assets is not a list.")
        return 2

    topic_id = topic_id_from_draft(draft)
    manifest_topic = manifest.get("topic_id")

    asset_results = []
    hard_failures = []
    rights_holds = []

    if topic_id != manifest_topic:
        hard_failures.append("MANIFEST_TOPIC_ID_MISMATCH")
    if not assets:
        hard_failures.append("MANIFEST_ASSET_LIST_EMPTY")
    warnings = []

    exact_hash_groups = {}
    visual_hashes = []
    direction_history = []

    for asset in assets:
        if not isinstance(asset, dict):
            continue

        asset_id = str(asset.get("asset_id", "UNKNOWN"))
        relative_path = str(asset.get("asset_path", ""))

        failures = []
        holds = []
        asset_warnings = []
        details = {}

        image_path, path_error = validate_asset_path(
            manifest_path.parent,
            relative_path,
        )

        if path_error:
            failures.append(path_error)

        else:
            try:
                file_hash = sha256_file(image_path)
                details["sha256"] = file_hash

                exact_hash_groups.setdefault(
                    file_hash, []
                ).append(asset_id)

                with Image.open(image_path) as image:
                    image.verify()

                with Image.open(image_path) as image:
                    width, height = image.size
                    image_format = image.format or ""

                details["dimensions"] = [width, height]
                details["image_format"] = image_format

                fmt = str(
                    asset.get("format", "landscape")
                ).strip().lower()

                minimum = MINIMUMS.get(fmt)

                if minimum is None:
                    failures.append("UNKNOWN_ASSET_FORMAT")
                else:
                    details["premium_minimum"] = list(minimum)

                    declared_minimum = asset.get(
                        "minimum_dimensions", {}
                    )

                    details["manifest_declared_minimum"] = {
                        "width": declared_minimum.get("width"),
                        "height": declared_minimum.get("height"),
                    }

                    if width < minimum[0] or height < minimum[1]:
                        failures.append(
                            f"BELOW_PREMIUM_MINIMUM_{minimum[0]}x{minimum[1]}"
                        )

                current_visual_hash = average_hash(image_path)

                for other_id, other_hash in visual_hashes:
                    score = hash_similarity(
                        current_visual_hash, other_hash
                    )

                    if score >= args.visual_warning_threshold:
                        asset_warnings.append({
                            "type": "PERCEPTUAL_SIMILARITY",
                            "other_asset_id": other_id,
                            "score": round(score, 4),
                        })

                visual_hashes.append(
                    (asset_id, current_visual_hash)
                )

            except (
                UnidentifiedImageError,
                OSError,
                ValueError,
            ) as exc:
                failures.append("IMAGE_UNREADABLE")
                details["image_error"] = str(exc)

        provenance_failures, provenance_holds = (
            validate_provenance(asset)
        )

        failures.extend(provenance_failures)
        holds.extend(provenance_holds)

        direction = normalize_text(
            str(asset.get("visual_direction", ""))
        )
        direction_grams = make_shingles(direction, 3)

        for other_id, other_grams in direction_history:
            score = jaccard_similarity(
                direction_grams, other_grams
            )

            if direction_grams and score >= 0.75:
                asset_warnings.append({
                    "type": "REPEATED_VISUAL_DIRECTION",
                    "other_asset_id": other_id,
                    "score": round(score, 4),
                })

        direction_history.append(
            (asset_id, direction_grams)
        )

        if failures:
            hard_failures.append(asset_id)

        if holds:
            rights_holds.append(asset_id)

        warnings.extend(
            f"{asset_id}:{item['type']}"
            for item in asset_warnings
        )

        asset_results.append({
            "asset_id": asset_id,
            "asset_path": relative_path,
            "source_type": (
                (asset.get("provenance") or {}).get("source_type")
            ),
            "details": details,
            "failures": sorted(set(failures)),
            "holds": sorted(set(holds)),
            "warnings": asset_warnings,
        })

    # Mark every member of an exact duplicate group except its
    # first occurrence. The original file is not modified.
    duplicate_groups = []

    for digest, ids in exact_hash_groups.items():
        if len(ids) > 1:
            duplicate_groups.append({
                "sha256": digest,
                "asset_ids": ids,
                "duplicate_asset_ids": ids[1:],
            })

            for duplicate_id in ids[1:]:
                for result in asset_results:
                    if result["asset_id"] == duplicate_id:
                        result["failures"].append(
                            "EXACT_DUPLICATE_OF_" + ids[0]
                        )
                        if duplicate_id not in hard_failures:
                            hard_failures.append(duplicate_id)

    # Script comparison: long-to-long and short-to-short,
    # excluding all earlier drafts belonging to the same topic.
    script_results = []
    script_holds = []

    for kind, filename in SCRIPT_NAMES.items():
        current_path = draft / filename
        current_text = load_script(current_path)

        if not current_text:
            script_holds.append(
                f"CURRENT_SCRIPT_MISSING_OR_EMPTY:{kind}"
            )
            script_results.append({
                "kind": kind,
                "status": "HOLD",
                "reason": "CURRENT_SCRIPT_MISSING_OR_EMPTY",
                "comparisons": [],
            })
            continue

        previous_scripts = collect_other_topic_scripts(
            runs_root, draft, kind
        )

        comparisons = []
        current_grams = make_shingles(current_text, 5)

        for previous in previous_scripts:
            old_text = previous["content"]
            old_grams = make_shingles(old_text, 5)

            jaccard_score = jaccard_similarity(
                current_grams, old_grams
            )

            sequence_score = SequenceMatcher(
                None,
                current_text[:15000],
                old_text[:15000],
            ).ratio()

            comparisons.append({
                "other_topic_id": previous["topic_id"],
                "file": str(previous["path"]),
                "jaccard_similarity": round(jaccard_score, 4),
                "sequence_similarity": round(sequence_score, 4),
            })

            if (
                jaccard_score >= args.script_jaccard_threshold
                or sequence_score >= args.script_sequence_threshold
            ):
                script_holds.append(
                    f"POSSIBLE_CROSS_TOPIC_SCRIPT_REUSE:{kind}:"
                    f"{previous['topic_id']}"
                )

        status = classify_script_corpus(
            previous_scripts,
            comparisons,
            args.script_jaccard_threshold,
            args.script_sequence_threshold,
        )
        if status == "BASELINE_SEED_NO_PRIOR_TOPIC_SCRIPTS":
            # No earlier in-project script exists to compare against.
            # This does not claim global or external originality.
            warnings.append(f"SCRIPT_CORPUS_BASELINE_SEED:{kind}")
        elif status == "HOLD_POSSIBLE_REUSE":
            script_holds.append(f"POSSIBLE_CROSS_TOPIC_SCRIPT_REUSE:{kind}")

        script_results.append({
            "kind": kind,
            "status": status,
            "current_script": str(current_path),
            "other_topic_comparison_count": len(previous_scripts),
            "comparisons": comparisons,
        })

    hard_failures = sorted(set(hard_failures))
    rights_holds = sorted(set(rights_holds))
    script_holds = sorted(set(script_holds))

    all_holds = sorted(set(rights_holds + script_holds))

    if hard_failures:
        gate_status = "FAIL_REGENERATION_REQUIRED"
    elif all_holds:
        gate_status = "HOLD_INSUFFICIENT_EVIDENCE"
    else:
        gate_status = "PASS_ORIGINALITY_GATE"

    failure_counts = Counter(
        failure
        for result in asset_results
        for failure in result["failures"]
    )

    hold_counts = Counter(
        hold
        for result in asset_results
        for hold in result["holds"]
    )

    report = {
        "gate_version": VERSION,
        "status": gate_status,
        "gate_pass": gate_status == "PASS_ORIGINALITY_GATE",
        "topic_id_from_path": topic_id,
        "topic_id_from_manifest": manifest_topic,
        "topic_id_match": topic_id == manifest_topic,
        "read_only": True,
        "source_manifest_modified": False,
        "source_assets_modified": False,
        "approval_flags_modified": False,
        "production_authorization": False,
        "render_authorization": False,
        "youtube_upload_authorization": False,
        "public_upload_authorization": False,
        "asset_count": len(asset_results),
        "failed_asset_ids": hard_failures,
        "rights_hold_asset_ids": rights_holds,
        "script_holds": script_holds,
        "script_corpus_baseline_seed": any(
            item.get("status") == "BASELINE_SEED_NO_PRIOR_TOPIC_SCRIPTS"
            for item in script_results
        ),
        "failure_counts": dict(failure_counts),
        "asset_hold_counts": dict(hold_counts),
        "duplicate_groups": duplicate_groups,
        "warnings": warnings,
        "warning_count": len(warnings),
        "assets": asset_results,
        "scripts": script_results,
        "thresholds": {
            "landscape": list(MINIMUMS["landscape"]),
            "vertical": list(MINIMUMS["vertical"]),
            "visual_similarity_warning": args.visual_warning_threshold,
            "script_jaccard_hold": args.script_jaccard_threshold,
            "script_sequence_hold": args.script_sequence_threshold,
        },
        "limitations": [
            "This is a technical and evidence gate, not a complete semantic originality detector.",
            "Perceptual hash and visual direction matches are warnings only.",
            "Script similarity is a heuristic and requires a growing cross-topic corpus.",
            "When no earlier project script exists, this first script seeds the project corpus; that does not prove global or external originality.",
            "No provider commercial terms are independently verified by this local gate.",
            "Rights uncertainty results in HOLD, not PASS.",
            "Project-authored local procedural assets require creator, code reference, rights basis, external-input exclusion and commercial-use declaration.",
            "A gate PASS never grants production or upload authorization.",
            "No source asset, manifest or approval record is modified.",
        ],
    }

    report_path = (
        draft / "premium_originality_gate_v1_3_report.json"
    )

    report_path.write_text(
        json.dumps(report, indent=2, ensure_ascii=False),
        encoding="utf-8",
    )

    print("=" * 64)
    print("PREMIUM ORIGINALITY GATE V1.3")
    print("=" * 64)
    print("STATUS:", gate_status)
    print("TOPIC:", topic_id)
    print("MANIFEST TOPIC MATCH:", topic_id == manifest_topic)
    print("ASSETS:", len(asset_results))
    print("FAILED ASSETS:", len(hard_failures))
    print("RIGHTS HOLD ASSETS:", len(rights_holds))
    print("SCRIPT HOLDS:", len(script_holds))
    print("EXACT DUPLICATE GROUPS:", len(duplicate_groups))
    print("WARNINGS:", len(warnings))

    print("\nFAILURE REASONS:")
    for reason, count in failure_counts.most_common():
        print(f"{reason}: {count}")

    print("\nRIGHTS HOLD REASONS:")
    for reason, count in hold_counts.most_common():
        print(f"{reason}: {count}")

    print("\nSCRIPT RESULTS:")
    for item in script_results:
        print(
            item["kind"],
            "|", item["status"],
            "| comparisons:", item.get(
                "other_topic_comparison_count", 0
            ),
        )

    print("\nSCRIPT HOLDS:")
    for reason in script_holds:
        print(reason)

    print("\nDUPLICATE GROUPS:")
    for group in duplicate_groups:
        print(", ".join(group["asset_ids"]))

    print("\nREPORT:", report_path)
    print("SOURCE MANIFEST MODIFIED: False")
    print("SOURCE ASSETS MODIFIED: False")
    print("APPROVAL FLAGS MODIFIED: False")
    print("PRODUCTION AUTHORIZATION: False")
    print("UPLOAD AUTHORIZATION: False")

    return 0 if gate_status == "PASS_ORIGINALITY_GATE" else 1

from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent

def validate_topic_id(topic):
    import re
    if not re.fullmatch(r"[A-Za-z0-9_-]{3,80}", str(topic or "")):
        raise ValueError("Invalid topic ID; path separators and traversal are not allowed.")

def latest_draft(topic):
    validate_topic_id(topic)
    base = ROOT / "production" / "runs" / topic / "drafts"
    if not base.is_dir():
        raise RuntimeError(f"Draft directory not found for topic: {topic}")
    candidates = []
    for p in base.iterdir():
        if p.is_dir() and (p / "creative_package.json").is_file():
            fact = p / "automated_fact_check_v2_report.json"
            if fact.is_file():
                try:
                    if json.loads(fact.read_text(encoding="utf-8-sig")).get("status") == "PASS":
                        candidates.append(p)
                except Exception:
                    pass
    if not candidates:
        raise RuntimeError("No draft with creative package and passing fact-check report.")
    return max(candidates, key=lambda p: p.stat().st_mtime)

def digest(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def seed_for(text, attempt):
    raw = (text + "|" + str(attempt)).encode("utf-8")
    return int(hashlib.sha256(raw).hexdigest()[:16], 16)

def make_artwork(task, output, attempt=0, run_salt=""):
    width, height = task["width"], task["height"]
    rng = random.Random(seed_for(task["asset_id"] + task["visual_direction"] + "|" + run_salt, attempt))

    palettes = [
        ((8, 15, 35), (24, 55, 86), (13, 112, 124)),
        ((18, 12, 35), (72, 35, 83), (169, 91, 82)),
        ((7, 24, 29), (18, 75, 78), (94, 137, 111)),
        ((22, 18, 15), (83, 57, 36), (169, 119, 66)),
        ((12, 17, 29), (39, 54, 83), (103, 89, 140)),
    ]
    c1, c2, accent = rng.choice(palettes)

    # Cinematic gradient background.
    gradient = Image.linear_gradient("L").resize((width, height))
    background = Image.new("RGB", (width, height), c1)
    top = Image.new("RGB", (width, height), c2)
    background = Image.composite(top, background, gradient)

    # Soft light source.
    glow = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    gd = ImageDraw.Draw(glow)
    gx = rng.randint(width // 4, width * 3 // 4)
    gy = rng.randint(height // 4, height * 3 // 4)
    radius = int(min(width, height) * rng.uniform(.20, .42))
    gd.ellipse((gx-radius, gy-radius, gx+radius, gy+radius),
               fill=(*accent, 115))
    glow = glow.filter(ImageFilter.GaussianBlur(radius * .55))
    background = Image.alpha_composite(background.convert("RGBA"), glow)

    # Original geometric documentary composition.
    layer = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    draw = ImageDraw.Draw(layer)
    short = min(width, height)

    # Fine grid and perspective lines.
    spacing = max(55, short // 13)
    for x in range(0, width, spacing):
        draw.line((x, 0, x, height), fill=(210, 225, 235, 13), width=1)
    for y in range(0, height, spacing):
        draw.line((0, y, width, y), fill=(210, 225, 235, 13), width=1)

    # Layered rings / orbital geometry.
    cx = rng.randint(width // 3, width * 2 // 3)
    cy = rng.randint(height // 3, height * 2 // 3)
    for k in range(5):
        r = int(short * (.12 + k * .075))
        draw.ellipse((cx-r, cy-r, cx+r, cy+r),
                     outline=(*accent, max(18, 90-k*12)),
                     width=max(2, short // 500))

    # Directional beams.
    for _ in range(16):
        x1 = rng.randint(-width // 3, width)
        y1 = rng.randint(-height // 3, height)
        x2 = x1 + rng.randint(-width // 2, width // 2)
        y2 = y1 + rng.randint(-height // 2, height // 2)
        draw.line((x1, y1, x2, y2),
                  fill=(*accent, rng.randint(12, 45)),
                  width=rng.randint(1, max(2, short // 300)))

    # Abstract focal forms.
    for _ in range(12):
        x = rng.randint(0, width)
        y = rng.randint(0, height)
        r = rng.randint(short // 45, short // 10)
        col = rng.choice([(*accent, 28), (220, 230, 240, 18), (4, 8, 18, 50)])
        draw.ellipse((x-r, y-r, x+r, y+r), fill=col,
                     outline=(*accent, 65), width=max(1, short // 600))

    # Unique scene-specific diagonal motif.
    points = [
        (rng.randint(0, width), rng.randint(0, height))
        for _ in range(7)
    ]
    draw.line(points, fill=(*accent, 150), width=max(3, short // 220),
              joint="curve")

    # Scene-aware visual motif. This is deterministic and uses only Pillow
    # primitives; it is not represented as photorealistic AI imagery.
    scene_text = " ".join([
        str(task.get("visual_direction", "")),
        str(task.get("narration_excerpt", "")),
        str(task.get("section", "")),
    ]).lower()
    cx, cy = width // 2, height // 2
    unit = max(12, short // 35)
    ink = (*accent, 220)
    pale = (225, 235, 245, 190)

    def line(coords, fill=ink, w=None):
        draw.line(coords, fill=fill, width=w or max(3, unit // 5), joint="curve")

    def ellipse(box, outline=ink, w=None, fill=None):
        draw.ellipse(box, outline=outline, width=w or max(3, unit // 5), fill=fill)

    if any(k in scene_text for k in ("money", "wealth", "finance", "cash", "debt", "invest", "rich", "poor", "price", "income")):
        # Coin stack + rising financial trajectory.
        for n in range(4):
            y = cy + unit * (2 - n)
            draw.rounded_rectangle((cx-unit*3, y, cx-unit, y+unit//2),
                radius=unit//5, outline=pale, width=max(3, unit//6), fill=(20, 35, 55, 120))
        line([(cx, cy+unit*2), (cx+unit, cy+unit), (cx+unit*2, cy+unit//2), (cx+unit*3, cy-unit*2)], w=max(5,unit//4))
        line([(cx+unit*2,cy-unit*2),(cx+unit*3,cy-unit*2),(cx+unit*3,cy-unit)], w=max(5,unit//4))
    elif any(k in scene_text for k in ("brain", "psychology", "mind", "thought", "emotion", "memory", "overthink", "behavior", "decision")):
        # Abstract brain hemispheres with neural connections.
        ellipse((cx-unit*3,cy-unit*3,cx,cy+unit*3), pale, max(4,unit//4))
        ellipse((cx,cy-unit*3,cx+unit*3,cy+unit*3), pale, max(4,unit//4))
        line([(cx,cy-unit*2),(cx-unit,cy-unit),(cx-unit*2,cy),(cx-unit,cy+unit),(cx,cy+unit*2)],w=max(3,unit//5))

    elif any(k in scene_text for k in ("power", "status", "hierarchy", "influence", "leader", "control", "authority", "dominance")):

    composed = Image.alpha_composite(background, layer)

    mask = Image.new("L", (width, height), 0)

def run(topic):
    validate_topic_id(topic)
    from research_approval_guard_v1 import require_research_approval
    require_research_approval(topic)
    draft = latest_draft(topic)
    creative_path = draft / "creative_package.json"
    creative = json.loads(creative_path.read_text(encoding="utf-8-sig"))
    tasks = collect_tasks(creative)
    if not tasks:
        raise RuntimeError("Creative package contains no visual tasks.")

    stamp = datetime.now().astimezone().strftime("%Y%m%d_%H%M%S")
    package_dir = draft / f"automated_visual_assets_v1_{stamp}"
    assets_dir = package_dir / "assets"
    assets_dir.mkdir(parents=True, exist_ok=False)

    results, hashes = [], set()
    for task in tasks:
        filename = task["asset_id"] + ".jpg"
        path = assets_dir / filename
        errors = []

        # Bounded local regeneration: maximum two attempts.
        selected_attempt = 0
        for attempt in range(2):
            selected_attempt = attempt
            make_artwork(task, path, attempt, run_salt=stamp)
            try:
                with Image.open(path) as im:
                    im.verify()
                with Image.open(path) as im:
                    valid = im.size == (task["width"], task["height"])
                file_hash = digest(path)
                if valid and file_hash not in hashes:
                    break
                errors.append("DIMENSION_OR_DUPLICATE_RETRY")
            except Exception as exc:
                errors.append("IMAGE_DECODE_RETRY:" + str(exc))
        else:
            results.append({
                **task, "asset_path": "assets/" + filename,
                "status": "REJECTED", "errors": errors
            })
            continue

        hashes.add(file_hash)
        results.append({
            **task,
            "asset_path": "assets/" + filename,
            "status": "AUTOMATED_TECHNICAL_PASS",
            "sha256": file_hash,
            "provenance": {
                 "source_type": "local_procedural_original",
                "creator": "Project-authored local visual renderer",
                "source_code_reference": "production/automated_visual_engine_v1.py",

                "commercial_use_confirmed": True,

                "external_images_used": False,
                "stock_media_used": False,
                "third_party_fonts_used": False,
                "generation_seed": seed_for(task["asset_id"] + task["visual_direction"] + "|" + stamp, selected_attempt),
                "generation_run_id": stamp,
            },
            "quality": {
                "width": task["width"], "height": task["height"],
                "format": "JPEG", "decode": "PASS",
                "exact_duplicate": False,
            },
            "errors": errors,
        })

    passed = sum(x["status"] == "AUTOMATED_TECHNICAL_PASS" for x in results)
    failed = len(results) - passed
    manifest = {
        "version": VERSION,
        "topic_id": topic,
        "created_at": datetime.now().astimezone().isoformat(),
        "draft_directory": str(draft),
        "package_directory": str(package_dir),
        "generation_method": "local_procedural_original",
        "external_media_used": False,
        "total_assets": len(results),
        "automated_technical_pass": passed,
        "rejected": failed,
        "assets": results,
        "automated_visual_gate": "PASS" if failed == 0 else "HOLD",
        "limitations": [
            "These are original procedural graphics, not photorealistic AI images.",
            "Automated checks validate dimensions, file integrity and exact duplicates.",
            "Visual semantic relevance is not independently understood by a vision model.",
            "No external stock or AI provider was used; this is not a legal opinion on copyright.",
            "Technical visual pass does not authorize rendering, production or publication.",
        ],
        "approval": {
            "automated_technical_gate_pass": failed == 0,
            "production_approved": False,
            "render_authorized": False,
            "upload_authorized": False,
            "public_upload_authorized": False,
        },
    }
    manifest_path = package_dir / "asset_manifest.json"
    manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")
    report_path = package_dir / "automated_visual_gate_report.json"
    report_path.write_text(json.dumps({

    print("PRODUCTION APPROVED: False")
    print("RENDER AUTHORIZED: False")
    print("UPLOAD AUTHORIZED: False")
    print("MANIFEST:", manifest_path)
    print("=" * 60)
    return 0 if failed == 0 else 1

from __future__ import annotations

import argparse
import hashlib
import json
import sys
from datetime import datetime
from pathlib import Path

VERSION = "1.0.0"


def read_json(path: Path):
    with path.open("r", encoding="utf-8-sig") as f:
        return json.load(f)


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def safe_child(root: Path, relative: str) -> Path:
    root = root.resolve()
    candidate = (root / relative).resolve()
    if candidate != root and root not in candidate.parents:
        raise ValueError(f"Asset path escapes package directory: {relative}")
    return candidate


def validate_reports(script_report_path: Path, visual_report_path: Path):
    errors = []
    warnings = []

    script = read_json(script_report_path)
    visual = read_json(visual_report_path)

    topic_id = script.get("topic_id")
    draft_dir = Path(script.get("draft_directory", "")).resolve()

    if not topic_id:
        errors.append("SCRIPT_TOPIC_ID_MISSING")

    if not draft_dir.is_dir():
        errors.append("DRAFT_DIRECTORY_MISSING")

    if script.get("status") != "SCRIPT_GATES_PASS_VISUAL_RENDER_PENDING":
        errors.append("SCRIPT_PIPELINE_STATUS_NOT_PASS")

    if script.get("fact_check_status") != "PASS":
        errors.append("FACT_CHECK_NOT_PASS")

    if script.get("script_qa_status") != "PASS":
        errors.append("SCRIPT_QA_NOT_PASS")

    if script.get("production_approved") is not False:
        errors.append("PRODUCTION_FLAG_NOT_LOCKED")

    if script.get("render_authorized") is not False:
        errors.append("SCRIPT_RENDER_FLAG_NOT_LOCKED")

    if script.get("upload_authorized") is not False:
        errors.append("SCRIPT_UPLOAD_FLAG_NOT_LOCKED")

    if visual.get("status") != "PASS":
        errors.append("VISUAL_REPORT_NOT_PASS")

    if visual.get("production_approved") is not False:
        errors.append("VISUAL_PRODUCTION_FLAG_NOT_LOCKED")

    if visual.get("render_authorized") is not False:
        errors.append("VISUAL_RENDER_FLAG_NOT_LOCKED")

    if visual.get("upload_authorized") is not False:
        errors.append("VISUAL_UPLOAD_FLAG_NOT_LOCKED")

    manifest_path = Path(visual.get("manifest", "")).resolve()

    if not manifest_path.is_file():
        errors.append("VISUAL_MANIFEST_MISSING")
        return script, visual, None, errors, warnings

    manifest = read_json(manifest_path)
    package_dir = manifest_path.parent

    if manifest.get("topic_id") != topic_id:
        errors.append("TOPIC_ID_MISMATCH")

    manifest_draft = Path(manifest.get("draft_directory", "")).resolve()
    if manifest_draft != draft_dir:
        errors.append("DRAFT_DIRECTORY_MISMATCH")

    if Path(manifest.get("package_directory", "")).resolve() != package_dir:
        errors.append("PACKAGE_DIRECTORY_MISMATCH")

    if visual_report_path.parent.resolve() != package_dir:
        errors.append("VISUAL_REPORT_PACKAGE_MISMATCH")

    if manifest.get("generation_method") != "local_procedural_original":
        warnings.append("UNEXPECTED_GENERATION_METHOD")

    if manifest.get("external_media_used") is not False:
        warnings.append("EXTERNAL_MEDIA_PROVENANCE_REQUIRES_REVIEW")

    assets = manifest.get("assets")
    if not isinstance(assets, list) or not assets:
        errors.append("ASSET_LIST_EMPTY_OR_INVALID")
        return script, visual, manifest, errors, warnings

    expected_total = manifest.get("total_assets")
    if expected_total != len(assets):
        errors.append("MANIFEST_ASSET_COUNT_MISMATCH")

    if visual.get("total") != len(assets):
        errors.append("VISUAL_REPORT_ASSET_COUNT_MISMATCH")

    if visual.get("passed") != len(assets):
        errors.append("VISUAL_REPORT_PASS_COUNT_MISMATCH")

    if visual.get("rejected") != 0:
        errors.append("VISUAL_REPORT_HAS_REJECTED_ASSETS")

    seen_hashes = set()
    seen_ids = set()
    verified = 0
    asset_errors = []

    try:
        from PIL import Image
    except Exception as exc:
        errors.append(f"PILLOW_UNAVAILABLE: {exc}")
        return script, visual, manifest, errors, warnings

    for index, asset in enumerate(assets, start=1):
        asset_id = asset.get("asset_id")
        rel_path = asset.get("asset_path")
        item_errors = []

        if not asset_id or asset_id in seen_ids:
            item_errors.append("MISSING_OR_DUPLICATE_ASSET_ID")
        else:
            seen_ids.add(asset_id)

        if asset.get("status") != "AUTOMATED_TECHNICAL_PASS":
            item_errors.append("ASSET_STATUS_NOT_PASS")

        if not rel_path:
            item_errors.append("ASSET_PATH_MISSING")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        try:
            path = safe_child(package_dir, rel_path)
        except Exception as exc:
            item_errors.append(f"UNSAFE_ASSET_PATH: {exc}")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        if not path.is_file():
            item_errors.append("ASSET_FILE_MISSING")
            asset_errors.append({"asset": asset_id, "errors": item_errors})
            continue

        actual_hash = sha256_file(path)
        declared_hash = asset.get("sha256")

        if not declared_hash or actual_hash != declared_hash:
            item_errors.append("SHA256_MISMATCH")

        if actual_hash in seen_hashes:
            item_errors.append("EXACT_DUPLICATE_ASSET")
        seen_hashes.add(actual_hash)

        expected_size = (
            (1920, 1080) if asset.get("format") == "landscape"
            else (1080, 1920) if asset.get("format") == "vertical"
            else None
        )

        try:
            with Image.open(path) as im:
                im.verify()

            with Image.open(path) as im:
                actual_size = im.size
                actual_format = im.format

            if expected_size is None:
                item_errors.append("UNKNOWN_ASSET_FORMAT")
            elif actual_size != expected_size:
                item_errors.append(
                    f"WRONG_DIMENSIONS: expected={expected_size}, actual={actual_size}"
                )

            if actual_format != "JPEG":
                item_errors.append(f"WRONG_IMAGE_FORMAT: {actual_format}")

        except Exception as exc:
            item_errors.append(f"IMAGE_DECODE_FAILED: {exc}")

        provenance = asset.get("provenance") or {}
        if provenance.get("source_type") != "local_procedural_original":
            item_errors.append("PROVENANCE_SOURCE_TYPE_INVALID")

        if provenance.get("external_images_used") is not False:
            item_errors.append("EXTERNAL_IMAGE_PROVENANCE_NOT_CLEAR")

        if provenance.get("stock_media_used") is not False:
            item_errors.append("STOCK_MEDIA_PROVENANCE_NOT_CLEAR")

        if not provenance.get("renderer"):
            item_errors.append("RENDERER_PROVENANCE_MISSING")

        if not provenance.get("rights_basis"):
            item_errors.append("RIGHTS_BASIS_MISSING")

        if not asset.get("visual_direction"):
            warnings.append(f"VISUAL_DIRECTION_MISSING:{asset_id}")

        if not asset.get("narration_excerpt") and asset.get("category") != "thumbnail":
            warnings.append(f"NARRATION_EXCERPT_MISSING:{asset_id}")

        if item_errors:
            asset_errors.append({"asset": asset_id, "errors": item_errors})
        else:
            verified += 1

    if len(seen_ids) != len(assets):
        errors.append("ASSET_ID_UNIQUENESS_FAILED")

    if len(seen_hashes) != len(assets):
        errors.append("ASSET_HASH_UNIQUENESS_FAILED")

    if verified != len(assets):
        errors.append("ONE_OR_MORE_ASSETS_FAILED_REVALIDATION")

    if asset_errors:
        errors.append("ASSET_VALIDATION_ERRORS_PRESENT")

    if script.get("public_upload_authorization") is not False:
        errors.append("PUBLIC_UPLOAD_AUTHORIZATION_NOT_LOCKED")

    approval = manifest.get("approval") or {}
    for key in (
        "production_approved",
        "render_authorized",
        "upload_authorized",
        "public_upload_authorized",
    ):
        if approval.get(key) is not False:
            errors.append(f"MANIFEST_APPROVAL_NOT_LOCKED:{key}")

    return script, visual, manifest, errors, warnings, {
        "verified_assets": verified,
        "asset_errors": asset_errors,
        "package_directory": str(package_dir),
        "manifest_path": str(manifest_path),
    }
